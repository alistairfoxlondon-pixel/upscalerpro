   136→   136→let workerWired = false;
   137→   137→const pending = new Map<string, { resolve: () => void }>();
   138→   138→const stallTimers = new Map<string, ReturnType<typeof setTimeout>>();
   139→   139→const STALL_TIMEOUT_MS = 300_000;
   140→   140→/** a fresh-context GPU retry after a Finalizing stall gets a shorter leash */
   141→   141→const GPU_RETRY_TIMEOUT_MS = 180_000;
   142→   142→let loopRunning = false;
   143→   143→
   144→   144→/* ---- patch-grid accumulation (flushed to the store at most ~8×/s) ---- */
   145→   145→const patchAccum = new Map<string, { cols: number; rows: number; done: number }>();
   146→   146→let patchFlushTimer: ReturnType<typeof setTimeout> | null = null;
   147→   147→
   148→   148→function flushPatches() {
   149→   149→  patchFlushTimer = null;
   150→   150→  if (!patchAccum.size) return;
   151→   151→  const entries = [...patchAccum.entries()];
   152→   152→  patchAccum.clear();
   153→   153→  useStore.setState((s) => ({
   154→   154→    items: s.items.map((it) => {
   155→   155→      const hit = entries.find(([id]) => id === it.id);
   156→   156→      return hit ? { ...it, patch: hit[1] } : it;
   157→   157→    }),
   158→   158→  }));
   159→   159→}
   160→   160→
   161→   161→function queuePatchFlush() {
   162→   162→  if (!patchFlushTimer) {
   163→   163→    patchFlushTimer = setTimeout(flushPatches, 120);
   164→   164→  }
   165→   165→}
   166→   166→
   167→   167→function onPatch(id: string, row: number, col: number, cols: number, rows: number) {
   168→   168→  if (!pending.has(id)) return;
   169→   169→  const prev = patchAccum.get(id);
   170→   170→  // a pass change (8× chain) swaps the grid dimensions — restart the count
   171→   171→  const base = prev && prev.cols === cols && prev.rows === rows ? prev : null;
   172→   172→  const done = Math.max(base?.done ?? 0, row * cols + col + 1);
   173→   173→  patchAccum.set(id, { cols, rows, done });
   174→   174→  queuePatchFlush();
   175→   175→}

   177→   177→function touchStall(id: string) {
   178→   178→  const prev = stallTimers.get(id);
   179→   179→  if (prev) clearTimeout(prev);
   180→   180→  const item = useStore.getState().items.find((i) => i.id === id);
   181→   181→  const timeout = item?.gpuRetry ? GPU_RETRY_TIMEOUT_MS : STALL_TIMEOUT_MS;
   182→   182→  stallTimers.set(
   183→   183→    id,
   184→   184→    setTimeout(() => {
   185→   185→      stallTimers.delete(id);
   186→   186→      onStall(id);
   187→   187→    }, timeout)
   188→   188→  );
   189→   189→}
   190→   190→
   191→   191→function clearStall(id: string) {
   192→   192→  const prev = stallTimers.get(id);
   193→   193→  if (prev) clearTimeout(prev);
   194→   194→  stallTimers.delete(id);
   195→   195→}
   196→   196→
   197→   197→async function onStall(id: string) {
   198→   198→  if (!pending.has(id)) return;
   199→   199→  const item = useStore.getState().items.find((i) => i.id === id);
   200→   200→  if (!item) return;
   201→   201→  const { toast } = await import('sonner');
   202→   202→  if (item.cpuRetry) {
   203→   203→    console.error('[pixelforge] processing stalled even on CPU fallback', id);
   204→   204→    patchItem(id, {
   205→   205→      status: 'error',
   206→   206→      phase: 'Failed',
   207→   207→      error: 'Processing stalled — try a smaller image or the Fast engine',
   208→   208→      result: undefined,
   209→   209→    });
   210→   210→    pending.get(id)?.resolve();
   211→   211→    pending.delete(id);
   212→   212→    return;
   213→   213→  }
   214→   214→  // The GPU readback is wedged — a cancel cannot unblock the worker's job
   215→   215→  // chain, so terminate it and give the retry a fresh worker.
   216→   216→  if (item.gpuRetry) {
   217→   217→    // Second stall — the fresh GPU context did not survive either. Fall back
   218→   218→    // to the pure-JS backend: slow but deterministic.
   219→   219→    toast.info('Graphics engine stalled again — retrying on CPU (slower but reliable)');
   220→   220→    resetWorker();
   221→   221→    patchItem(id, {
   222→   222→      cpuRetry: true,
   223→   223→      gpuRetry: false,
   224→   224→      status: 'queued',
   225→   225→      phase: 'Queued (CPU retry)',
   226→   226→      progress: 0,
   227→   227→      result: undefined,
   228→   228→    });
   229→   229→  } else if (/^Finalizing/i.test(item.phase)) {
   230→   230→    // The AI pass already finished — only the GPU→CPU readback is suspect.
   231→   231→    // A fresh WebGL context is usually healthy again, and re-running the AI
   232→   232→    // pass is far cheaper than a full CPU inference (which can take minutes).
   233→   233→    toast.info('Graphics engine stalled during output — retrying on a fresh GPU context');
   234→   234→    resetWorker();
   235→   235→    patchItem(id, {
   236→   236→      gpuRetry: true,
   237→   237→      status: 'queued',
   238→   238→      phase: 'Queued (GPU retry)',
   239→   239→      progress: 0,
   240→   240→      result: undefined,
   241→   241→    });
   242→   242→  } else {
   243→   243→    // Compute-phase stall — go straight to the reliable CPU fallback.
   244→   244→    toast.info('Graphics engine stalled — restarting worker and retrying on CPU (slower but reliable)');
   245→   245→    resetWorker();
   246→   246→    patchItem(id, {
   247→   247→      cpuRetry: true,
   248→   248→      gpuRetry: false,
   249→   249→      status: 'queued',
   250→   250→      phase: 'Queued (CPU retry)',
   251→   251→      progress: 0,
   252→   252→      result: undefined,
   253→   253→    });
   254→   254→  }
   255→   255→  pending.get(id)?.resolve();
   256→   256→  pending.delete(id);
   257→   257→}

   259→   259→function patchItem(id: string, patch: Partial<QueueItem>) {
   260→   260→  useStore.setState((s) => ({
   261→   261→    items: s.items.map((it) => (it.id === id ? { ...it, ...patch } : it)),
   262→   262→  }));
   263→   263→}

   265→   265→/**
   266→   266→ * Persist a finished result to IndexedDB so it can be restored after a
   267→   267→ * reload. The original is re-encoded to a capped preview; thumbnails are
   268→   268→ * already small. Never throws.
   269→   269→ */
   270→   270→async function savePersisted(item: QueueItem, blob: Blob, result: ResultData) {
   271→   271→  try {
   272→   272→    const [thumbBlob, originalBlob] = await Promise.all([
   273→   273→      fetch(item.thumbUrl).then((r) => r.blob()).catch(() => null),
   274→   274→      item.originalUrl
   275→   275→        ? makePreviewUrlFromUrl(item.originalUrl).catch(() => null)
   276→   276→        : Promise.resolve(null),
   277→   277→    ]);
   278→   278→    if (!thumbBlob || !originalBlob) return;
   279→   279→    const data: PersistedResult = {
   280→   280→      w: item.w,
   281→   281→      h: item.h,
   282→   282→      sizeIn: item.sizeIn,
   283→   283→      name: item.name,
   284→   284→      mime: item.mime,
   285→   285→      thumb: thumbBlob,
   286→   286→      original: originalBlob,
   287→   287→      result: {
   288→   288→        blob,
   289→   289→        w: result.w,
   290→   290→        h: result.h,
   291→   291→        ms: result.ms,
   292→   292→        scale: result.scale,
   293→   293→        preset: result.preset,
   294→   294→        format: result.format,
   295→   295→        clamped: result.clamped,
   296→   296→      },
   297→   297→      savedAt: Date.now(),
   298→   298→    };
   299→   299→    await idbPutResult(item.id, data);
   300→   300→  } catch {
   301→   301→    /* best-effort */
   302→   302→  }
   303→   303→}

   305→   305→/** Re-encodes any image object-URL to a ≤2048px JPEG for compact storage. */
   306→   306→async function makePreviewUrlFromUrl(url: string, maxSide = 2048): Promise<Blob> {
   307→   307→  const src = await createImageBitmap(await (await fetch(url)).blob());
   308→   308→  try {
   309→   309→    const ratio = Math.min(1, maxSide / Math.max(src.width, src.height));
   310→   310→    const w = Math.max(1, Math.round(src.width * ratio));
   311→   311→    const h = Math.max(1, Math.round(src.height * ratio));
   312→   312→    const canvas = document.createElement('canvas');
   313→   313→    canvas.width = w;
   314→   314→    canvas.height = h;
   315→   315→    const ctx = canvas.getContext('2d');
   316→   316→    if (!ctx) throw new Error('Canvas unavailable');
   317→   317→    ctx.imageSmoothingQuality = 'high';
   318→   318→    ctx.drawImage(src, 0, 0, w, h);
   319→   319→    const blob = await new Promise<Blob | null>((res) => canvas.toBlob(res, 'image/jpeg', 0.9));
   320→   320→    if (!blob) throw new Error('Preview encoding failed');
   321→   321→    return blob;
   322→   322→  } finally {
   323→   323→    src.close();
   324→   324→  }
   325→   325→}

   327→   327→function wireWorker() {
   328→   328→  if (workerWired) return;
   329→   329→  workerWired = true;
   330→   330→  onWorkerMessage((m: WorkerOutMessage) => {
   331→   331→    const state = useStore.getState();
   332→   332→    switch (m.type) {
   333→   333→      case 'backend':
   334→   334→        useStore.setState({ backend: m.backend, gpu: m.renderer ?? null });
   335→   335→        break;
   336→   336→      case 'debug':
   337→   337→        console.info('[pf-worker]', m.text);
   338→   338→        break;
   339→   339→      case 'phase':
   340→   340→        if (pending.has(m.id)) {
   341→   341→          touchStall(m.id);
   342→   342→          patchItem(m.id, { phase: m.phase });
   343→   343→        }
   344→   344→        break;
   345→   345→      case 'model-status': {
   346→   346→        const key = `${m.preset}:${m.scale}`;
   347→   347→        if (state.modelKey === key || m.status === 'error') {
   348→   348→          useStore.setState({ modelStatus: m.status, modelMessage: m.message ?? null });
   349→   349→        }
   350→   350→        if (m.status === 'loading' && state.activeId) {
   351→   351→          patchItem(state.activeId, { phase: 'Loading AI model' });
   352→   352→          touchStall(state.activeId);
   353→   353→        }
   354→   354→        if (m.status === 'ready' && state.activeId) {
   355→   355→          const active = state.items.find((i) => i.id === state.activeId);
   356→   356→          if (active && active.phase === 'Loading AI model') {
   357→   357→            patchItem(active.id, { phase: 'Upscaling' });
   358→   358→          }
   359→   359→        }
   360→   360→        break;
   361→   361→      }
   362→   362→      case 'progress':
   363→   363→        touchStall(m.id);
   364→   364→        patchItem(m.id, {
   365→   365→          progress: 0.02 + m.rate * 0.93,
   366→   366→          phase: m.rate >= 0.999 ? 'Finalizing' : 'Upscaling',
   367→   367→        });
   368→   368→        break;
   369→   369→      case 'patch':
   370→   370→        onPatch(m.id, m.row, m.col, m.cols, m.rows);
   371→   371→        break;
   372→   372→      case 'done': {
   373→   373→        if (!pending.has(m.id)) break; // stale message (item re-queued or removed)
   374→   374→        clearStall(m.id);
   375→   375→        patchAccum.delete(m.id);
   376→   376→        const item = state.items.find((i) => i.id === m.id);
   377→   377→        const url = URL.createObjectURL(m.blob);
   378→   378→        const planned = item?.result;
   379→   379→        const result: ResultData = {
   380→   380→          url,
   381→   381→          size: m.blob.size,
   382→   382→          w: m.width,
   383→   383→          h: m.height,
   384→   384→          ms: m.ms,
   385→   385→          scale: (planned?.scale ?? useStore.getState().settings.scale) as ScaleFactor,
   386→   386→          preset: (planned?.preset ?? useStore.getState().settings.preset) as PresetId,
   387→   387→          format: m.blob.type.includes('jpeg')
   388→   388→            ? 'jpeg'
   389→   389→            : m.blob.type.includes('webp')
   390→   390→              ? 'webp'
   391→   391→              : 'png',
   392→   392→          clamped: planned?.clamped ?? false,
   393→   393→        };
   394→   394→        patchItem(m.id, {
   395→   395→          status: 'done',
   396→   396→          progress: 1,
   397→   397→          phase: 'Done',
   398→   398→          result,
   399→   399→          patch: null,
   400→   400→        });
   401→   401→        useStore.setState((s) => ({
   402→   402→          totals: {
   403→   403→            images: s.totals.images + 1,
   404→   404→            pixelsIn: s.totals.pixelsIn + (item ? item.w * item.h : 0),
   405→   405→            pixelsOut: s.totals.pixelsOut + m.width * m.height,
   406→   406→            ms: s.totals.ms + m.ms,
   407→   407→            bytesIn: s.totals.bytesIn + (item?.sizeIn ?? 0),
   408→   408→            bytesOut: s.totals.bytesOut + m.blob.size,
   409→   409→          },
   410→   410→        }));
   411→   411→        pending.get(m.id)?.resolve();
   412→   412→        pending.delete(m.id);
   413→   413→        // persist the finished result so it survives a reload (best-effort)
   414→   414→        if (item) void savePersisted(item, m.blob, result);
   415→   415→        break;
   416→   416→      }
   417→   417→      case 'error': {
   418→   418→        if (!pending.has(m.id)) break; // stale message (item re-queued or removed)
   419→   419→        clearStall(m.id);
   420→   420→        patchAccum.delete(m.id);
   421→   421→        const canceled = /^canceled$/i.test(m.message);
   422→   422→        patchItem(m.id, {
   423→   423→          status: canceled ? 'canceled' : 'error',
   424→   424→          phase: canceled ? 'Canceled' : 'Failed',
   425→   425→          error: canceled ? undefined : m.message,
   426→   426→          progress: 0,
   427→   427→          result: undefined,
   428→   428→        });
   429→   429→        if (!canceled) {
   430→   430→          console.error('[pixelforge]', m.message);
   431→   431→          void import('sonner').then(({ toast }) =>
   432→   432→            toast.error('Upscaling failed', { description: m.message })
   433→   433→          );
   434→   434→        }
   435→   435→        pending.get(m.id)?.resolve();
   436→   436→        pending.delete(m.id);
   437→   437→        break;
   438→   438→      }
   439→   439→    }
   440→   440→  });
   441→   441→}

   443→   443→async function processItem(id: string): Promise<void> {
   444→   444→  const state = useStore.getState();
   445→   445→  const item = state.items.find((i) => i.id === id);
   446→   446→  if (!item) return;
   447→   447→  if (!item.file) {
   448→   448→    // restored items have no source file — treat as informational failure
   449→   449→    patchItem(id, {
   450→   450→      status: 'error',
   451→   451→      phase: 'Failed',
   452→   452→      error: 'Original file is not kept after a reload — add the image again',
   453→   453→    });
   454→   454→    return;
   455→   455→  }
   456→   456→  const { preset, scale, format, jpegQuality, denoise, sharpen, keepExif } = state.settings;
   457→   457→
   458→   458→  patchItem(id, {
   459→   459→    status: 'processing',
   460→   460→    progress: 0,
   461→   461→    phase: 'Preparing',
   462→   462→    error: undefined,
   463→   463→    patch: null,
   464→   464→  });
   465→   465→  useStore.setState({ activeId: id });
   466→   466→
   467→   467→  const resolved: OutputFormat = format === 'auto' ? resolveFormat(item.mime, item.name) : format;
   468→   468→
   469→   469→  // memory safety: pick the largest allowed scale
   470→   470→  const fit = resolveScale(item.w, item.h, scale);
   471→   471→  if (!fit) {
   472→   472→    patchItem(id, {
   473→   473→      status: 'error',
   474→   474→      phase: 'Failed',
   475→   475→      error: 'Image is too large to upscale on this device',
   476→   476→    });
   477→   477→    return;
   478→   478→  }
   479→   479→
   480→   480→  try {
   481→   481→    const { bitmap } = await decodeImageFile(item.file);
   482→   482→    const modelKey = `${preset}:${fit.scale}`;
   483→   483→    useStore.setState({ modelKey, modelStatus: null, modelMessage: null });
   484→   484→    patchItem(id, {
   485→   485→      phase: 'Loading model',
   486→   486→      progress: 0.01,
   487→   487→      // planned result placeholder — carries the plan (scale/format) and
   488→   488→      // estimated output size until the worker reports the real values
   489→   489→      result: {
   490→   490→        url: '',
   491→   491→        size: 0,
   492→   492→        w: item.w * fit.scale,
   493→   493→        h: item.h * fit.scale,
   494→   494→        ms: 0,
   495→   495→        scale: fit.scale,
   496→   496→        preset,
   497→   497→        format: resolved,
   498→   498→        clamped: fit.clamped,
   499→   499→      },
   500→   500→    });
   501→   501→    // Opt-in metadata preservation: pull the original JPEG APP1 EXIF segment
   502→   502→    // (first 256 KB header scan — cheap) so the worker can splice it into the
   503→   503→    // encoded output. Only meaningful for JPEG results.
   504→   504→    const exif =
   505→   505→      keepExif && resolved === 'jpeg' ? await extractExifSegment(item.file) : null;
   506→   506→    const done = new Promise<void>((resolve) => pending.set(id, { resolve }));
   507→   507→    touchStall(id);
   508→   508→    const transfer: Transferable[] = [bitmap];
   509→   509→    if (exif) transfer.push(exif);
   510→   510→    postToWorker(
   511→   511→      {
   512→   512→        type: 'upscale',
   513→   513→        id,
   514→   514→        bitmap,
   515→   515→        scale: fit.scale,
   516→   516→        preset,
   517→   517→        format: resolved,
   518→   518→        quality: jpegQuality,
   519→   519→        denoise,
   520→   520→        sharpen,
   521→   521→        exif: exif ?? undefined,
   522→   522→        backendHint: item.cpuRetry ? 'cpu' : undefined,
   523→   523→      },
   524→   524→      transfer
   525→   525→    );
   526→   526→    await done;
   527→   527→    clearStall(id);
   528→   528→  } catch (err) {
   529→   529→    patchItem(id, {
   530→   530→      status: 'error',
   531→   531→      phase: 'Failed',
   532→   532→      error: err instanceof Error ? err.message : 'Could not decode image',
   533→   533→      result: undefined,
   534→   534→    });
   535→   535→  }
   536→   536→}
   537→   537→
   538→   538→async function ensureLoop() {
   539→   539→  if (loopRunning) return;
   540→   540→  loopRunning = true;
   541→   541→  useStore.setState({ busy: true });
   542→   542→  let completed = 0;
   543→   543→  try {
   544→   544→    for (;;) {
   545→   545→      const s = useStore.getState();
   546→   546→      if (s.paused) break;
   547→   547→      const next = s.items.find((i) => i.status === 'queued');
   548→   548→      if (!next) break;
   549→   549→      await processItem(next.id);
   550→   550→      const after = useStore.getState().items.find((i) => i.id === next.id);
   551→   551→      if (after?.status === 'done') completed++;
   552→   552→    }
   553→   553→  } finally {
   554→   554→    loopRunning = false;
   555→   555→    useStore.setState({ busy: false, activeId: null });
   556→   556→    // batch-complete feedback with a one-tap ZIP export
   557→   557→    if (completed > 0 && !useStore.getState().paused) {
   558→   558→      const { toast } = await import('sonner');
   559→   559→      toast.success(completed === 1 ? '1 image ready' : `${completed} images ready`, {
   560→   560→        description: 'Processed entirely on this device.',
   561→   561→        action: { label: 'Download ZIP', onClick: () => void downloadAllAsZip() },
   562→   562→        duration: 10_000,
   563→   563→      });
   564→   564→    }
   565→   565→  }
   566→   566→}

   820→   820→export function downloadBlob(blob: Blob, filename: string) {
   821→   821→  const url = URL.createObjectURL(blob);
   822→   822→  const a = document.createElement('a');
   823→   823→  a.href = url;
   824→   824→  a.download = filename;
   825→   825→  document.body.appendChild(a);
   826→   826→  a.click();
   827→   827→  a.remove();
   828→   828→  setTimeout(() => URL.revokeObjectURL(url), 4000);
   829→   829→}

   836→   836→export async function downloadAllAsZip() {
   837→   837→  const { items } = useStore.getState();
   838→   838→  const done = items.filter((i) => i.status === 'done' && i.result && i.zip !== false);
   839→   839→  if (!done.length) return;
   840→   840→  const { toast } = await import('sonner');
   841→   841→  const t = toast.loading(`Packing ${done.length} image${done.length > 1 ? 's' : ''}…`);
   842→   842→  try {
   843→   843→    const JSZip = (await import('jszip')).default;
   844→   844→    const zip = new JSZip();
   845→   845→    const used = new Set<string>();
   846→   846→    for (const item of done) {
   847→   847→      const r = item.result!;
   848→   848→      let filename = resultFilename(item.name, r.scale, r.format);
   849→   849→      let i = 1;
   850→   850→      while (used.has(filename)) {
   851→   851→        filename = resultFilename(item.name, r.scale, r.format).replace(/(\.\w+)$/, `-${i++}$1`);
   852→   852→      }
   853→   853→      used.add(filename);
   854→   854→      const blob = await (await fetch(r.url)).blob();
   855→   855→      zip.file(filename, blob);
   856→   856→    }
   857→   857→    const out = await zip.generateAsync({ type: 'blob' });
   858→   858→    const stamp = new Date().toISOString().slice(0, 16).replace(/[-:T]/g, '');
   859→   859→    downloadBlob(out, `pixelforge-upscaled-${stamp}.zip`);
   860→   860→    toast.success(`ZIP ready — ${formatBytes(out.size)}`, { id: t });
   861→   861→  } catch (err) {
   862→   862→    console.error(err);
   863→   863→    toast.error('Could not build the ZIP archive', { id: t });
   864→   864→  }
   865→   865→}
