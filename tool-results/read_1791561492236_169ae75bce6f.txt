   130→   130→let workerWired = false;
   131→   131→const pending = new Map<string, { resolve: () => void }>();
   132→   132→const stallTimers = new Map<string, ReturnType<typeof setTimeout>>();
   133→   133→const STALL_TIMEOUT_MS = 300_000;
   134→   134→let loopRunning = false;
   135→   135→
   136→   136→/* ---- patch-grid accumulation (flushed to the store at most ~8×/s) ---- */
   137→   137→const patchAccum = new Map<string, { cols: number; rows: number; done: number }>();
   138→   138→let patchFlushTimer: ReturnType<typeof setTimeout> | null = null;
   139→   139→
   140→   140→function flushPatches() {
   141→   141→  patchFlushTimer = null;
   142→   142→  if (!patchAccum.size) return;
   143→   143→  const entries = [...patchAccum.entries()];
   144→   144→  patchAccum.clear();
   145→   145→  useStore.setState((s) => ({
   146→   146→    items: s.items.map((it) => {
   147→   147→      const hit = entries.find(([id]) => id === it.id);
   148→   148→      return hit ? { ...it, patch: hit[1] } : it;
   149→   149→    }),
   150→   150→  }));
   151→   151→}
   152→   152→
   153→   153→function queuePatchFlush() {
   154→   154→  if (!patchFlushTimer) {
   155→   155→    patchFlushTimer = setTimeout(flushPatches, 120);
   156→   156→  }
   157→   157→}
   158→   158→
   159→   159→function onPatch(id: string, row: number, col: number, cols: number, rows: number) {
   160→   160→  if (!pending.has(id)) return;
   161→   161→  const prev = patchAccum.get(id);
   162→   162→  // a pass change (8× chain) swaps the grid dimensions — restart the count
   163→   163→  const base = prev && prev.cols === cols && prev.rows === rows ? prev : null;
   164→   164→  const done = Math.max(base?.done ?? 0, row * cols + col + 1);
   165→   165→  patchAccum.set(id, { cols, rows, done });
   166→   166→  queuePatchFlush();
   167→   167→}
   168→   168→
   169→   169→function touchStall(id: string) {
   170→   170→  const prev = stallTimers.get(id);
   171→   171→  if (prev) clearTimeout(prev);
   172→   172→  stallTimers.set(
   173→   173→    id,
   174→   174→    setTimeout(() => {
   175→   175→      stallTimers.delete(id);
   176→   176→      onStall(id);
   177→   177→    }, STALL_TIMEOUT_MS)
   178→   178→  );
   179→   179→}
   180→   180→
   181→   181→function clearStall(id: string) {
   182→   182→  const prev = stallTimers.get(id);
   183→   183→  if (prev) clearTimeout(prev);
   184→   184→  stallTimers.delete(id);
   185→   185→}
   186→   186→
   187→   187→async function onStall(id: string) {
   188→   188→  if (!pending.has(id)) return;
   189→   189→  const item = useStore.getState().items.find((i) => i.id === id);
   190→   190→  if (!item) return;
   191→   191→  const { toast } = await import('sonner');
   192→   192→  if (item.cpuRetry) {
   193→   193→    console.error('[pixelforge] processing stalled even on CPU fallback', id);
   194→   194→    patchItem(id, {
   195→   195→      status: 'error',
   196→   196→      phase: 'Failed',
   197→   197→      error: 'Processing stalled — try a smaller image or the Fast engine',
   198→   198→      result: undefined,
   199→   199→    });
   200→   200→    pending.get(id)?.resolve();
   201→   201→    pending.delete(id);
   202→   202→    return;
   203→   203→  }
   204→   204→  // The GPU readback is wedged — a cancel cannot unblock the worker's job
   205→   205→  // chain, so terminate it and give the retry a fresh worker + CPU backend.
   206→   206→  toast.info('Graphics engine stalled — restarting worker and retrying on CPU (slower but reliable)');
   207→   207→  resetWorker();
   208→   208→  patchItem(id, {
   209→   209→    cpuRetry: true,
   210→   210→    status: 'queued',
   211→   211→    phase: 'Queued (CPU retry)',
   212→   212→    progress: 0,
   213→   213→    result: undefined,
   214→   214→  });
   215→   215→  pending.get(id)?.resolve();
   216→   216→  pending.delete(id);
   217→   217→}
   218→   218→
   219→   219→function patchItem(id: string, patch: Partial<QueueItem>) {
   220→   220→  useStore.setState((s) => ({
   221→   221→    items: s.items.map((it) => (it.id === id ? { ...it, ...patch } : it)),
   222→   222→  }));
   223→   223→}

   225→   225→/**
   226→   226→ * Persist a finished result to IndexedDB so it can be restored after a
   227→   227→ * reload. The original is re-encoded to a capped preview; thumbnails are
   228→   228→ * already small. Never throws.
   229→   229→ */
   230→   230→async function savePersisted(item: QueueItem, blob: Blob, result: ResultData) {
   231→   231→  try {
   232→   232→    const [thumbBlob, originalBlob] = await Promise.all([
   233→   233→      fetch(item.thumbUrl).then((r) => r.blob()).catch(() => null),
   234→   234→      item.originalUrl
   235→   235→        ? makePreviewUrlFromUrl(item.originalUrl).catch(() => null)
   236→   236→        : Promise.resolve(null),
   237→   237→    ]);
   238→   238→    if (!thumbBlob || !originalBlob) return;
   239→   239→    const data: PersistedResult = {
   240→   240→      w: item.w,
   241→   241→      h: item.h,
   242→   242→      sizeIn: item.sizeIn,
   243→   243→      name: item.name,
   244→   244→      mime: item.mime,
   245→   245→      thumb: thumbBlob,
   246→   246→      original: originalBlob,
   247→   247→      result: {
   248→   248→        blob,
   249→   249→        w: result.w,
   250→   250→        h: result.h,
   251→   251→        ms: result.ms,
   252→   252→        scale: result.scale,
   253→   253→        preset: result.preset,
   254→   254→        format: result.format,
   255→   255→        clamped: result.clamped,
   256→   256→      },
   257→   257→      savedAt: Date.now(),
   258→   258→    };
   259→   259→    await idbPutResult(item.id, data);
   260→   260→  } catch {
   261→   261→    /* best-effort */
   262→   262→  }
   263→   263→}

   265→   265→/** Re-encodes any image object-URL to a ≤2048px JPEG for compact storage. */
   266→   266→async function makePreviewUrlFromUrl(url: string, maxSide = 2048): Promise<Blob> {
   267→   267→  const src = await createImageBitmap(await (await fetch(url)).blob());
   268→   268→  try {
   269→   269→    const ratio = Math.min(1, maxSide / Math.max(src.width, src.height));
   270→   270→    const w = Math.max(1, Math.round(src.width * ratio));
   271→   271→    const h = Math.max(1, Math.round(src.height * ratio));
   272→   272→    const canvas = document.createElement('canvas');
   273→   273→    canvas.width = w;
   274→   274→    canvas.height = h;
   275→   275→    const ctx = canvas.getContext('2d');
   276→   276→    if (!ctx) throw new Error('Canvas unavailable');
   277→   277→    ctx.imageSmoothingQuality = 'high';
   278→   278→    ctx.drawImage(src, 0, 0, w, h);
   279→   279→    const blob = await new Promise<Blob | null>((res) => canvas.toBlob(res, 'image/jpeg', 0.9));
   280→   280→    if (!blob) throw new Error('Preview encoding failed');
   281→   281→    return blob;
   282→   282→  } finally {
   283→   283→    src.close();
   284→   284→  }
   285→   285→}

   287→   287→function wireWorker() {
   288→   288→  if (workerWired) return;
   289→   289→  workerWired = true;
   290→   290→  onWorkerMessage((m: WorkerOutMessage) => {
   291→   291→    const state = useStore.getState();
   292→   292→    switch (m.type) {
   293→   293→      case 'backend':
   294→   294→        useStore.setState({ backend: m.backend, gpu: m.renderer ?? null });
   295→   295→        break;
   296→   296→      case 'debug':
   297→   297→        console.info('[pf-worker]', m.text);
   298→   298→        break;
   299→   299→      case 'phase':
   300→   300→        if (pending.has(m.id)) {
   301→   301→          touchStall(m.id);
   302→   302→          patchItem(m.id, { phase: m.phase });
   303→   303→        }
   304→   304→        break;
   305→   305→      case 'model-status': {
   306→   306→        const key = `${m.preset}:${m.scale}`;
   307→   307→        if (state.modelKey === key || m.status === 'error') {
   308→   308→          useStore.setState({ modelStatus: m.status, modelMessage: m.message ?? null });
   309→   309→        }
   310→   310→        if (m.status === 'loading' && state.activeId) {
   311→   311→          patchItem(state.activeId, { phase: 'Loading AI model' });
   312→   312→          touchStall(state.activeId);
   313→   313→        }
   314→   314→        if (m.status === 'ready' && state.activeId) {
   315→   315→          const active = state.items.find((i) => i.id === state.activeId);
   316→   316→          if (active && active.phase === 'Loading AI model') {
   317→   317→            patchItem(active.id, { phase: 'Upscaling' });
   318→   318→          }
   319→   319→        }
   320→   320→        break;
   321→   321→      }
   322→   322→      case 'progress':
   323→   323→        touchStall(m.id);
   324→   324→        patchItem(m.id, {
   325→   325→          progress: 0.02 + m.rate * 0.93,
   326→   326→          phase: m.rate >= 0.999 ? 'Finalizing' : 'Upscaling',
   327→   327→        });
   328→   328→        break;
   329→   329→      case 'patch':
   330→   330→        onPatch(m.id, m.row, m.col, m.cols, m.rows);
   331→   331→        break;
   332→   332→      case 'done': {
   333→   333→        if (!pending.has(m.id)) break; // stale message (item re-queued or removed)
   334→   334→        clearStall(m.id);
   335→   335→        patchAccum.delete(m.id);
   336→   336→        const item = state.items.find((i) => i.id === m.id);
   337→   337→        const url = URL.createObjectURL(m.blob);
   338→   338→        const planned = item?.result;
   339→   339→        const result: ResultData = {
   340→   340→          url,
   341→   341→          size: m.blob.size,
   342→   342→          w: m.width,
   343→   343→          h: m.height,
   344→   344→          ms: m.ms,
   345→   345→          scale: (planned?.scale ?? useStore.getState().settings.scale) as ScaleFactor,
   346→   346→          preset: (planned?.preset ?? useStore.getState().settings.preset) as PresetId,
   347→   347→          format: m.blob.type.includes('jpeg')
   348→   348→            ? 'jpeg'
   349→   349→            : m.blob.type.includes('webp')
   350→   350→              ? 'webp'
   351→   351→              : 'png',
   352→   352→          clamped: planned?.clamped ?? false,
   353→   353→        };
   354→   354→        patchItem(m.id, {
   355→   355→          status: 'done',
   356→   356→          progress: 1,
   357→   357→          phase: 'Done',
   358→   358→          result,
   359→   359→          patch: null,
   360→   360→        });
   361→   361→        useStore.setState((s) => ({
   362→   362→          totals: {
   363→   363→            images: s.totals.images + 1,
   364→   364→            pixelsIn: s.totals.pixelsIn + (item ? item.w * item.h : 0),
   365→   365→            pixelsOut: s.totals.pixelsOut + m.width * m.height,
   366→   366→            ms: s.totals.ms + m.ms,
   367→   367→            bytesIn: s.totals.bytesIn + (item?.sizeIn ?? 0),
   368→   368→            bytesOut: s.totals.bytesOut + m.blob.size,
   369→   369→          },
   370→   370→        }));
   371→   371→        pending.get(m.id)?.resolve();
   372→   372→        pending.delete(m.id);
   373→   373→        // persist the finished result so it survives a reload (best-effort)
   374→   374→        if (item) void savePersisted(item, m.blob, result);
   375→   375→        break;
   376→   376→      }
   377→   377→      case 'error': {
   378→   378→        if (!pending.has(m.id)) break; // stale message (item re-queued or removed)
   379→   379→        clearStall(m.id);
   380→   380→        patchAccum.delete(m.id);
   381→   381→        const canceled = /^canceled$/i.test(m.message);
   382→   382→        patchItem(m.id, {
   383→   383→          status: canceled ? 'canceled' : 'error',
   384→   384→          phase: canceled ? 'Canceled' : 'Failed',
   385→   385→          error: canceled ? undefined : m.message,
   386→   386→          progress: 0,
   387→   387→          result: undefined,
   388→   388→        });
   389→   389→        if (!canceled) {
   390→   390→          console.error('[pixelforge]', m.message);
   391→   391→          void import('sonner').then(({ toast }) =>
   392→   392→            toast.error('Upscaling failed', { description: m.message })
   393→   393→          );
   394→   394→        }
   395→   395→        pending.get(m.id)?.resolve();
   396→   396→        pending.delete(m.id);
   397→   397→        break;
   398→   398→      }
   399→   399→    }
   400→   400→  });
   401→   401→}

   403→   403→async function processItem(id: string): Promise<void> {
   404→   404→  const state = useStore.getState();
   405→   405→  const item = state.items.find((i) => i.id === id);
   406→   406→  if (!item) return;
   407→   407→  if (!item.file) {
   408→   408→    // restored items have no source file — treat as informational failure
   409→   409→    patchItem(id, {
   410→   410→      status: 'error',
   411→   411→      phase: 'Failed',
   412→   412→      error: 'Original file is not kept after a reload — add the image again',
   413→   413→    });
   414→   414→    return;
   415→   415→  }
   416→   416→  const { preset, scale, format, jpegQuality, denoise, sharpen } = state.settings;
   417→   417→
   418→   418→  patchItem(id, {
   419→   419→    status: 'processing',
   420→   420→    progress: 0,
   421→   421→    phase: 'Preparing',
   422→   422→    error: undefined,
   423→   423→    patch: null,
   424→   424→  });
   425→   425→  useStore.setState({ activeId: id });
   426→   426→
   427→   427→  const resolved: OutputFormat = format === 'auto' ? resolveFormat(item.mime, item.name) : format;
   428→   428→
   429→   429→  // memory safety: pick the largest allowed scale
   430→   430→  const fit = resolveScale(item.w, item.h, scale);
   431→   431→  if (!fit) {
   432→   432→    patchItem(id, {
   433→   433→      status: 'error',
   434→   434→      phase: 'Failed',
   435→   435→      error: 'Image is too large to upscale on this device',
   436→   436→    });
   437→   437→    return;
   438→   438→  }
   439→   439→
   440→   440→  try {
   441→   441→    const { bitmap } = await decodeImageFile(item.file);
   442→   442→    const modelKey = `${preset}:${fit.scale}`;
   443→   443→    useStore.setState({ modelKey, modelStatus: null, modelMessage: null });
   444→   444→    patchItem(id, {
   445→   445→      phase: 'Loading model',
   446→   446→      progress: 0.01,
   447→   447→      // planned result placeholder — carries the plan (scale/format) and
   448→   448→      // estimated output size until the worker reports the real values
   449→   449→      result: {
   450→   450→        url: '',
   451→   451→        size: 0,
   452→   452→        w: item.w * fit.scale,
   453→   453→        h: item.h * fit.scale,
   454→   454→        ms: 0,
   455→   455→        scale: fit.scale,
   456→   456→        preset,
   457→   457→        format: resolved,
   458→   458→        clamped: fit.clamped,
   459→   459→      },
   460→   460→    });
   461→   461→    const done = new Promise<void>((resolve) => pending.set(id, { resolve }));
   462→   462→    touchStall(id);
   463→   463→    postToWorker(
   464→   464→      {
   465→   465→        type: 'upscale',
   466→   466→        id,
   467→   467→        bitmap,
   468→   468→        scale: fit.scale,
   469→   469→        preset,
   470→   470→        format: resolved,
   471→   471→        quality: jpegQuality,
   472→   472→        denoise,
   473→   473→        sharpen,
   474→   474→        backendHint: item.cpuRetry ? 'cpu' : undefined,
   475→   475→      },
   476→   476→      [bitmap]
   477→   477→    );
   478→   478→    await done;
   479→   479→    clearStall(id);
   480→   480→  } catch (err) {
   481→   481→    patchItem(id, {
   482→   482→      status: 'error',
   483→   483→      phase: 'Failed',
   484→   484→      error: err instanceof Error ? err.message : 'Could not decode image',
   485→   485→      result: undefined,
   486→   486→    });
   487→   487→  }
   488→   488→}

   490→   490→async function ensureLoop() {
   491→   491→  if (loopRunning) return;
   492→   492→  loopRunning = true;
   493→   493→  useStore.setState({ busy: true });
   494→   494→  let completed = 0;
   495→   495→  try {
   496→   496→    for (;;) {
   497→   497→      const s = useStore.getState();
   498→   498→      if (s.paused) break;
   499→   499→      const next = s.items.find((i) => i.status === 'queued');

   501→   501→      await processItem(next.id);
   502→   502→      const after = useStore.getState().items.find((i) => i.id === next.id);
   503→   503→      if (after?.status === 'done') completed++;
   504→   504→    }
   505→   505→  } finally {
   506→   506→    loopRunning = false;
   507→   507→    useStore.setState({ busy: false, activeId: null });
   508→   508→    // batch-complete feedback with a one-tap ZIP export
   509→   509→    if (completed > 0 && !useStore.getState().paused) {
   510→   510→      const { toast } = await import('sonner');
   511→   511→      toast.success(completed === 1 ? '1 image ready' : `${completed} images ready`, {
   512→   512→        description: 'Processed entirely on this device.',
   513→   513→        action: { label: 'Download ZIP', onClick: () => void downloadAllAsZip() },
   514→   514→        duration: 10_000,
   515→   515→      });
   516→   516→    }
   517→   517→  }
   518→   518→}

   771→   771→export function downloadBlob(blob: Blob, filename: string) {
   772→   772→  const url = URL.createObjectURL(blob);
   773→   773→  const a = document.createElement('a');
   774→   774→  a.href = url;
   775→   775→  a.download = filename;
   776→   776→  document.body.appendChild(a);
   777→   777→  a.click();
   778→   778→  a.remove();
   779→   779→  setTimeout(() => URL.revokeObjectURL(url), 4000);
   780→   780→}

   783→   783→  const base = name.replace(/\.[^.]+$/, '') || 'image';

   787→   787→export async function downloadAllAsZip() {
   788→   788→  const { items } = useStore.getState();
   789→   789→  const done = items.filter((i) => i.status === 'done' && i.result && i.zip !== false);
   790→   790→  if (!done.length) return;
   791→   791→  const { toast } = await import('sonner');
   792→   792→  const t = toast.loading(`Packing ${done.length} image${done.length > 1 ? 's' : ''}…`);
   793→   793→  try {
   794→   794→    const JSZip = (await import('jszip')).default;
   795→   795→    const zip = new JSZip();
   796→   796→    const used = new Set<string>();
   797→   797→    for (const item of done) {
   798→   798→      const r = item.result!;
   799→   799→      let filename = resultFilename(item.name, r.scale, r.format);
   800→   800→      let i = 1;
   801→   801→      while (used.has(filename)) {
   802→   802→        filename = resultFilename(item.name, r.scale, r.format).replace(/(\.\w+)$/, `-${i++}$1`);
   803→   803→      }
   804→   804→      used.add(filename);
   805→   805→      const blob = await (await fetch(r.url)).blob();
   806→   806→      zip.file(filename, blob);
   807→   807→    }
   808→   808→    const out = await zip.generateAsync({ type: 'blob' });
   809→   809→    const stamp = new Date().toISOString().slice(0, 16).replace(/[-:T]/g, '');
   810→   810→    downloadBlob(out, `pixelforge-upscaled-${stamp}.zip`);
   811→   811→    toast.success(`ZIP ready — ${formatBytes(out.size)}`, { id: t });
   812→   812→  } catch (err) {
   813→   813→    console.error(err);
   814→   814→    toast.error('Could not build the ZIP archive', { id: t });
   815→   815→  }
   816→   816→}
