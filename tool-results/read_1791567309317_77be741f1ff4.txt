   147→let workerWired = false;
   148→const pending = new Map<string, { resolve: () => void }>();
   149→const stallTimers = new Map<string, ReturnType<typeof setTimeout>>();
   150→const STALL_TIMEOUT_MS = 300_000;
   151→/** a fresh-context GPU retry after a Finalizing stall gets a shorter leash */
   152→const GPU_RETRY_TIMEOUT_MS = 180_000;
   153→let loopRunning = false;
   154→
   155→/* ---- patch-grid accumulation (flushed to the store at most ~8×/s) ---- */
   156→const patchAccum = new Map<string, { cols: number; rows: number; done: number }>();
   157→let patchFlushTimer: ReturnType<typeof setTimeout> | null = null;
   158→
   159→function flushPatches() {
   160→  patchFlushTimer = null;
   161→  if (!patchAccum.size) return;
   162→  const entries = [...patchAccum.entries()];
   163→  patchAccum.clear();
   164→  useStore.setState((s) => ({
   165→    items: s.items.map((it) => {
   166→      const hit = entries.find(([id]) => id === it.id);
   167→      return hit ? { ...it, patch: hit[1] } : it;
   168→    }),
   169→  }));
   170→}
   171→
   172→function queuePatchFlush() {
   173→  if (!patchFlushTimer) {
   174→    patchFlushTimer = setTimeout(flushPatches, 120);
   175→  }
   176→}
   177→
   178→function onPatch(id: string, row: number, col: number, cols: number, rows: number) {
   179→  if (!pending.has(id)) return;
   180→  const prev = patchAccum.get(id);
   181→  // a pass change (8× chain) swaps the grid dimensions — restart the count
   182→  const base = prev && prev.cols === cols && prev.rows === rows ? prev : null;
   183→  const done = Math.max(base?.done ?? 0, row * cols + col + 1);
   184→  patchAccum.set(id, { cols, rows, done });
   185→  queuePatchFlush();
   186→}
   187→
   188→function touchStall(id: string) {
   189→  const prev = stallTimers.get(id);
   190→  if (prev) clearTimeout(prev);
   191→  const item = useStore.getState().items.find((i) => i.id === id);
   192→  const timeout = item?.gpuRetry ? GPU_RETRY_TIMEOUT_MS : STALL_TIMEOUT_MS;
   193→  stallTimers.set(
   194→    id,
   195→    setTimeout(() => {
   196→      stallTimers.delete(id);
   197→      onStall(id);
   198→    }, timeout)
   199→  );
   200→}
   201→
   202→function clearStall(id: string) {
   203→  const prev = stallTimers.get(id);
   204→  if (prev) clearTimeout(prev);
   205→  stallTimers.delete(id);
   206→}
   207→
   208→async function onStall(id: string) {
   209→  if (!pending.has(id)) return;
   210→  const item = useStore.getState().items.find((i) => i.id === id);
   211→  if (!item) return;
   212→  const { toast } = await import('sonner');
   213→  if (item.cpuRetry) {
   214→    console.error('[pixelforge] processing stalled even on CPU fallback', id);
   215→    patchItem(id, {
   216→      status: 'error',
   217→      phase: 'Failed',
   218→      error: 'Processing stalled — try a smaller image or the Fast engine',
   219→      result: undefined,
   220→    });
   221→    pending.get(id)?.resolve();
   222→    pending.delete(id);
   223→    return;
   224→  }
   225→  // The GPU readback is wedged — a cancel cannot unblock the worker's job
   226→  // chain, so terminate it and give the retry a fresh worker.
   227→  if (item.gpuRetry) {
   228→    // Second stall — the fresh GPU context did not survive either. Fall back
   229→    // to the pure-JS backend: slow but deterministic.
   230→    toast.info('Graphics engine stalled again — retrying on CPU (slower but reliable)');
   231→    resetWorker();
   232→    patchItem(id, {
   233→      cpuRetry: true,
   234→      gpuRetry: false,
   235→      status: 'queued',
   236→      phase: 'Queued (CPU retry)',
   237→      progress: 0,
   238→      result: undefined,
   239→    });
   240→  } else if (/^Finalizing/i.test(item.phase)) {
   241→    // The AI pass already finished — only the GPU→CPU readback is suspect.
   242→    // A fresh WebGL context is usually healthy again, and re-running the AI
   243→    // pass is far cheaper than a full CPU inference (which can take minutes).
   244→    toast.info('Graphics engine stalled during output — retrying on a fresh GPU context');
   245→    resetWorker();
   246→    patchItem(id, {
   247→      gpuRetry: true,
   248→      status: 'queued',
   249→      phase: 'Queued (GPU retry)',
   250→      progress: 0,
   251→      result: undefined,
   252→    });
   253→  } else {
   254→    // Compute-phase stall — go straight to the reliable CPU fallback.
   255→    toast.info('Graphics engine stalled — restarting worker and retrying on CPU (slower but reliable)');
   256→    resetWorker();
   257→    patchItem(id, {
   258→      cpuRetry: true,
   259→      gpuRetry: false,
   260→      status: 'queued',
   261→      phase: 'Queued (CPU retry)',
   262→      progress: 0,
   263→      result: undefined,
   264→    });
   265→  }
   266→  pending.get(id)?.resolve();
   267→  pending.delete(id);
   268→}

   270→function patchItem(id: string, patch: Partial<QueueItem>) {
   271→  useStore.setState((s) => ({
   272→    items: s.items.map((it) => (it.id === id ? { ...it, ...patch } : it)),
   273→  }));
   274→}

   276→/**
   277→ * Persist a finished result to IndexedDB so it can be restored after a
   278→ * reload. The original is re-encoded to a capped preview; thumbnails are
   279→ * already small. Never throws.
   280→ */
   281→async function savePersisted(item: QueueItem, blob: Blob, result: ResultData) {
   282→  try {
   283→    const [thumbBlob, originalBlob] = await Promise.all([
   284→      fetch(item.thumbUrl).then((r) => r.blob()).catch(() => null),
   285→      item.originalUrl
   286→        ? makePreviewUrlFromUrl(item.originalUrl).catch(() => null)
   287→        : Promise.resolve(null),
   288→    ]);
   289→    if (!thumbBlob || !originalBlob) return;
   290→    const data: PersistedResult = {
   291→      w: item.w,
   292→      h: item.h,
   293→      sizeIn: item.sizeIn,
   294→      name: item.name,
   295→      mime: item.mime,
   296→      thumb: thumbBlob,
   297→      original: originalBlob,
   298→      result: {
   299→        blob,
   300→        w: result.w,
   301→        h: result.h,
   302→        ms: result.ms,
   303→        scale: result.scale,
   304→        preset: result.preset,
   305→        format: result.format,
   306→        clamped: result.clamped,
   307→        target: result.target,
   308→      },
   309→      savedAt: Date.now(),
   310→    };
   311→    await idbPutResult(item.id, data);
   312→  } catch {
   313→    /* best-effort */
   314→  }
   315→}

   317→/** Re-encodes any image object-URL to a ≤2048px JPEG for compact storage. */
   318→async function makePreviewUrlFromUrl(url: string, maxSide = 2048): Promise<Blob> {
   319→  const src = await createImageBitmap(await (await fetch(url)).blob());
   320→  try {
   321→    const ratio = Math.min(1, maxSide / Math.max(src.width, src.height));
   322→    const w = Math.max(1, Math.round(src.width * ratio));
   323→    const h = Math.max(1, Math.round(src.height * ratio));
   324→    const canvas = document.createElement('canvas');
   325→    canvas.width = w;
   326→    canvas.height = h;
   327→    const ctx = canvas.getContext('2d');
   328→    if (!ctx) throw new Error('Canvas unavailable');
   329→    ctx.imageSmoothingQuality = 'high';
   330→    ctx.drawImage(src, 0, 0, w, h);
   331→    const blob = await new Promise<Blob | null>((res) => canvas.toBlob(res, 'image/jpeg', 0.9));
   332→    if (!blob) throw new Error('Preview encoding failed');
   333→    return blob;
   334→  } finally {
   335→    src.close();
   336→  }
   337→}

   339→function wireWorker() {
   340→  if (workerWired) return;
   341→  workerWired = true;
   342→  onWorkerMessage((m: WorkerOutMessage) => {
   343→    const state = useStore.getState();
   344→    switch (m.type) {
   345→      case 'backend':
   346→        useStore.setState({ backend: m.backend, gpu: m.renderer ?? null });
   347→        break;
   348→      case 'debug':
   349→        console.info('[pf-worker]', m.text);
   350→        break;
   351→      case 'phase':
   352→        if (pending.has(m.id)) {
   353→          touchStall(m.id);
   354→          patchItem(m.id, { phase: m.phase });
   355→        }
   356→        break;
   357→      case 'model-status': {
   358→        const key = `${m.preset}:${m.scale}`;
   359→        if (state.modelKey === key || m.status === 'error') {
   360→          useStore.setState({ modelStatus: m.status, modelMessage: m.message ?? null });
   361→        }
   362→        if (m.status === 'loading' && state.activeId) {
   363→          patchItem(state.activeId, { phase: 'Loading AI model' });
   364→          touchStall(state.activeId);
   365→        }
   366→        if (m.status === 'ready' && state.activeId) {
   367→          const active = state.items.find((i) => i.id === state.activeId);
   368→          if (active && active.phase === 'Loading AI model') {
   369→            patchItem(active.id, { phase: 'Upscaling' });
   370→          }
   371→        }
   372→        break;
   373→      }
   374→      case 'progress':
   375→        touchStall(m.id);
   376→        patchItem(m.id, {
   377→          progress: 0.02 + m.rate * 0.93,
   378→          phase: m.rate >= 0.999 ? 'Finalizing' : 'Upscaling',
   379→        });
   380→        break;
   381→      case 'patch':
   382→        onPatch(m.id, m.row, m.col, m.cols, m.rows);
   383→        break;
   384→      case 'done': {
   385→        if (!pending.has(m.id)) break; // stale message (item re-queued or removed)
   386→        clearStall(m.id);
   387→        patchAccum.delete(m.id);
   388→        const item = state.items.find((i) => i.id === m.id);
   389→        const url = URL.createObjectURL(m.blob);
   390→        const planned = item?.result;
   391→        const result: ResultData = {
   392→          url,
   393→          size: m.blob.size,
   394→          w: m.width,
   395→          h: m.height,
   396→          ms: m.ms,
   397→          scale: (planned?.scale ?? useStore.getState().settings.scale) as ScaleFactor,
   398→          preset: (planned?.preset ?? useStore.getState().settings.preset) as PresetId,
   399→          format: m.blob.type.includes('jpeg')
   400→            ? 'jpeg'
   401→            : m.blob.type.includes('webp')
   402→              ? 'webp'
   403→              : 'png',
   404→          clamped: planned?.clamped ?? false,
   405→          target: planned?.target,
   406→        };
   407→        patchItem(m.id, {
   408→          status: 'done',
   409→          progress: 1,
   410→          phase: 'Done',
   411→          result,
   412→          patch: null,
   413→        });
   414→        useStore.setState((s) => ({
   415→          totals: {
   416→            images: s.totals.images + 1,
   417→            pixelsIn: s.totals.pixelsIn + (item ? item.w * item.h : 0),
   418→            pixelsOut: s.totals.pixelsOut + m.width * m.height,
   419→            ms: s.totals.ms + m.ms,
   420→            bytesIn: s.totals.bytesIn + (item?.sizeIn ?? 0),
   421→            bytesOut: s.totals.bytesOut + m.blob.size,
   422→          },
   423→        }));
   424→        pending.get(m.id)?.resolve();
   425→        pending.delete(m.id);
   426→        // persist the finished result so it survives a reload (best-effort)
   427→        if (item) void savePersisted(item, m.blob, result);
   428→        break;
   429→      }
   430→      case 'error': {
   431→        if (!pending.has(m.id)) break; // stale message (item re-queued or removed)
   432→        clearStall(m.id);
   433→        patchAccum.delete(m.id);
   434→        const canceled = /^canceled$/i.test(m.message);
   435→        patchItem(m.id, {
   436→          status: canceled ? 'canceled' : 'error',
   437→          phase: canceled ? 'Canceled' : 'Failed',
   438→          error: canceled ? undefined : m.message,
   439→          progress: 0,
   440→          result: undefined,
   441→        });
   442→        if (!canceled) {
   443→          console.error('[pixelforge]', m.message);
   444→          void import('sonner').then(({ toast }) =>
   445→            toast.error('Upscaling failed', { description: m.message })
   446→          );
   447→        }
   448→        pending.get(m.id)?.resolve();
   449→        pending.delete(m.id);
   450→        break;
   451→      }
   452→    }
   453→  });
   454→}

   456→async function processItem(id: string): Promise<void> {
   457→  const state = useStore.getState();
   458→  const item = state.items.find((i) => i.id === id);
   459→  if (!item) return;
   460→  if (!item.file) {
   461→    // restored items have no source file — treat as informational failure
   462→    patchItem(id, {
   463→      status: 'error',
   464→      phase: 'Failed',
   465→      error: 'Original file is not kept after a reload — add the image again',
   466→    });
   467→    return;
   468→  }
   469→  const { preset, scale, format, jpegQuality, denoise, sharpen, keepExif } = state.settings;
   470→
   471→  patchItem(id, {
   472→    status: 'processing',
   473→    progress: 0,
   474→    phase: 'Preparing',
   475→    error: undefined,
   476→    patch: null,
   477→  });
   478→  useStore.setState({ activeId: id });
   479→
   480→  const resolved: OutputFormat = format === 'auto' ? resolveFormat(item.mime, item.name) : format;
   481→
   482→  // memory safety: pick the largest allowed scale (factor mode) or the smallest
   483→  // AI scale that meets the target size, then resize (target mode)
   484→  const fit: SizeFit | null =
   485→    state.settings.scaleMode === 'target'
   486→      ? resolveScaleForTarget(item.w, item.h, state.settings.targetSide)
   487→      : resolveScale(item.w, item.h, scale);
   488→  if (!fit || fit.error) {
   489→    patchItem(id, {
   490→      status: 'error',
   491→      phase: 'Failed',
   492→      error: fit?.error ?? 'Image is too large to upscale on this device',
   493→    });
   494→    return;
   495→  }
   496→
   497→  try {
   498→    const { bitmap } = await decodeImageFile(item.file);
   499→    const modelKey = `${preset}:${fit.scale}`;
   500→    useStore.setState({ modelKey, modelStatus: null, modelMessage: null });
   501→    // planned result placeholder — carries the plan (scale/format) and
   502→    // estimated output size until the worker reports the real values
   503→    const planned: ResultData = {
   504→      url: '',
   505→      size: 0,
   506→      w: fit.outW,
   507→      h: fit.outH,
   508→      ms: 0,
   509→      scale: fit.scale,
   510→      preset,
   511→      format: resolved,
   512→      clamped: fit.clamped ?? false,
   513→      target:
   514→        state.settings.scaleMode === 'target' && !fit.short && !fit.error
   515→          ? state.settings.targetSide
   516→          : undefined,
   517→    };
   518→    patchItem(id, {
   519→      phase: 'Loading model',
   520→      progress: 0.01,
   521→      result: planned,
   522→    });
   523→    // Opt-in metadata preservation: pull the original JPEG APP1 EXIF segment
   524→    // (first 256 KB header scan — cheap) so the worker can splice it into the
   525→    // encoded output. Only meaningful for JPEG results.
   526→    const exif =
   527→      keepExif && resolved === 'jpeg' ? await extractExifSegment(item.file) : null;
   528→    const done = new Promise<void>((resolve) => pending.set(id, { resolve }));
   529→    touchStall(id);
   530→    const transfer: Transferable[] = [bitmap];
   531→    if (exif) transfer.push(exif);
   532→    postToWorker(
   533→      {
   534→        type: 'upscale',
   535→        id,
   536→        bitmap,
   537→        scale: fit.scale,
   538→        preset,
   539→        format: resolved,
   540→        quality: jpegQuality,
   541→        denoise,
   542→        sharpen,
   543→        exif: exif ?? undefined,
   544→        targetSide: planned.target,
   545→        backendHint: item.cpuRetry ? 'cpu' : undefined,
   546→      },
   547→      transfer
   548→    );
   549→    await done;
   550→    clearStall(id);
   551→  } catch (err) {
   552→    patchItem(id, {
   553→      status: 'error',
   554→      phase: 'Failed',
   555→      error: err instanceof Error ? err.message : 'Could not decode image',
   556→      result: undefined,
   557→    });
   558→  }
   559→}

   561→async function ensureLoop() {
   562→  if (loopRunning) return;
   563→  loopRunning = true;
   564→  useStore.setState({ busy: true });
   565→  let completed = 0;
   566→  try {
   567→    for (;;) {
   568→      const s = useStore.getState();
   569→      if (s.paused) break;
   570→      const next = s.items.find((i) => i.status === 'queued');
   571→      if (!next) break;
   572→      await processItem(next.id);
   573→      const after = useStore.getState().items.find((i) => i.id === next.id);
   574→      if (after?.status === 'done') completed++;
   575→    }
   576→  } finally {
   577→    loopRunning = false;
   578→    useStore.setState({ busy: false, activeId: null });
   579→    // batch-complete feedback with a one-tap ZIP export
   580→    if (completed > 0 && !useStore.getState().paused) {
   581→      const { toast } = await import('sonner');
   582→      toast.success(completed === 1 ? '1 image ready' : `${completed} images ready`, {
   583→        description: 'Processed entirely on this device.',
   584→        action: { label: 'Download ZIP', onClick: () => void downloadAllAsZip() },
   585→        duration: 10_000,
   586→      });
   587→    }
   588→  }
   589→}

   592→
   593→const initialTotals: Totals = {
   594→  images: 0,
   595→  pixelsIn: 0,
   596→  pixelsOut: 0,
   597→  ms: 0,
   598→  bytesIn: 0,
   599→  bytesOut: 0,
   600→};
   601→
   602→export const useStore = create<UpscalerState>()(
   603→  persist(
   604→    (set, get) => ({
   605→  items: [],
   606→  settings: { ...DEFAULT_SETTINGS },
   607→  busy: false,
   608→  paused: false,
   609→  activeId: null,
   610→  backend: null,
   611→  gpu: null,
   612→  modelKey: null,
   613→  modelStatus: null,
   614→  modelMessage: null,
   615→  totals: initialTotals,
   616→  compareId: null,
   617→
   618→  addFiles: async (files) => {
   619→    wireWorker();
   620→    const list = files.filter(
   621→      (f) =>
   622→        f.size > 0 &&
   623→        (f.type.startsWith('image/') || /\.(heic|heif|hif|tif|tiff|avif|webp|bmp)$/i.test(f.name))
   624→    );
   625→    if (!list.length) {
   626→      const { toast } = await import('sonner');
   627→      toast.error('No supported images found', {
   628→        description: 'Drop JPEG, PNG, WebP, AVIF, GIF, BMP, HEIC or TIFF files.',
   629→      });
   630→      return;
   631→    }
   632→    const capped = list.slice(0, MAX_BATCH_FILES);
   633→    // duplicate guard: same name + byte size already queued (or twice in this
   634→    // batch) — a re-added file would just waste GPU time on an identical job
   635→    const seen = new Set<string>();
   636→    for (const it of useStore.getState().items) {
   637→      if (it.file) seen.add(`${it.name}:${it.sizeIn}`);
   638→    }
   639→    let dupes = 0;
   640→    const items: QueueItem[] = [];
   641→    for (const file of capped) {
   642→      const key = `${file.name || 'image'}:${file.size}`;
   643→      if (seen.has(key)) {
   644→        dupes++;
   645→        continue;
   646→      }
   647→      seen.add(key);
   648→      const id = crypto.randomUUID();
   649→      const name = file.name || `image-${id.slice(0, 6)}`;
   650→      try {
   651→        const { bitmap, renderable } = await decodeImageFile(file);
   652→        const pixels = bitmap.width * bitmap.height;
   653→        if (pixels > MAX_INPUT_PIXELS) {
   654→          bitmap.close();
   655→          items.push({
   656→            id, file, name, mime: file.type || 'image/*', sizeIn: file.size,
   657→            w: 0, h: 0, thumbUrl: '', originalUrl: '',
   658→            status: 'error', progress: 0, phase: 'Failed',
   659→            error: `Image is ${Math.round(pixels / 1e6)}MP — max input is ${Math.round(MAX_INPUT_PIXELS / 1e6)}MP`,
   660→          });
   661→          continue;
   662→        }
   663→        const thumbUrl = await makeThumbUrl(bitmap);
   664→        const originalUrl = renderable ? URL.createObjectURL(file) : await makePreviewUrl(bitmap);
   665→        const w = bitmap.width;
   666→        const h = bitmap.height;
   667→        bitmap.close();
   668→        items.push({
   669→          id, file, name, mime: file.type || 'image/*', sizeIn: file.size,
   670→          w, h, thumbUrl, originalUrl,
   671→          status: 'queued', progress: 0, phase: 'Queued',
   672→        });
   673→      } catch (err) {
   674→        items.push({
   675→          id, file, name, mime: file.type || 'image/*', sizeIn: file.size,
   676→          w: 0, h: 0, thumbUrl: '', originalUrl: '',
   677→          status: 'error', progress: 0, phase: 'Failed',
   678→          error: err instanceof Error ? err.message : 'Could not decode image',
   679→        });
   680→      }
   681→    }
   682→    set((s) => ({ items: [...s.items, ...items] }));
   683→    if (dupes > 0) {
   684→      const { toast } = await import('sonner');
   685→      toast.info(`${dupes} duplicate file${dupes > 1 ? 's' : ''} skipped`, {
   686→        description: 'Already in the queue.',
   687→      });
   688→    }
   689→    const skipped = list.length - capped.length;
   690→    if (skipped > 0) {
   691→      const { toast } = await import('sonner');
   692→      toast.warning(`${skipped} file${skipped > 1 ? 's' : ''} skipped`, {
   693→        description: `Batch limit is ${MAX_BATCH_FILES} images at a time.`,
   694→      });
   695→    }
   696→    void ensureLoop();
   697→  },
   698→
   699→  addSample: async (kind: SampleKind = 'photo') => {
   700→    const { createSampleFile } = await import('./utils');
   701→    try {
   702→      const file = await createSampleFile(kind);
   703→      await get().addFiles([file]);
   704→    } catch (err) {
   705→      const { toast } = await import('sonner');
   706→      toast.error('Could not create the sample image');
   707→      console.error(err);
   708→    }
   709→  },
   710→
   711→  removeItem: (id) => {
   712→    const s = get();
   713→    const item = s.items.find((i) => i.id === id);
   714→    if (!item) return;
   715→    if (item.status === 'processing') {
   716→      postToWorker({ type: 'cancel', id });
   717→    }
   718→    if (item.thumbUrl) URL.revokeObjectURL(item.thumbUrl);
   719→    if (item.originalUrl) URL.revokeObjectURL(item.originalUrl);
   720→    if (item.result?.url) URL.revokeObjectURL(item.result.url);
   721→    void idbDeleteResult(id);
   722→    set((st) => ({
   723→      items: st.items.filter((i) => i.id !== id),
   724→      compareId: st.compareId === id ? null : st.compareId,
   725→    }));
   726→  },
   727→
   728→  clearFinished: () => {
   729→    const s = get();
   730→    for (const item of s.items) {
   731→      if (item.status === 'done' || item.status === 'canceled' || item.status === 'error') {
   732→        if (item.thumbUrl) URL.revokeObjectURL(item.thumbUrl);
   733→        if (item.originalUrl) URL.revokeObjectURL(item.originalUrl);
   734→        if (item.result?.url) URL.revokeObjectURL(item.result.url);
   735→        void idbDeleteResult(item.id);
   736→      }
   737→    }
   738→    set((st) => ({
   739→      items: st.items.filter(
   740→        (i) => i.status === 'queued' || i.status === 'processing'
   741→      ),
   742→      compareId: null,
   743→    }));
   744→  },
   745→
   746→  clearAll: () => {
   747→    const s = get();
   748→    for (const item of s.items) {
   749→      if (item.status === 'processing') {
   750→        postToWorker({ type: 'cancel', id: item.id });
   751→      }
   752→      if (item.thumbUrl) URL.revokeObjectURL(item.thumbUrl);
   753→      if (item.originalUrl) URL.revokeObjectURL(item.originalUrl);
   754→      if (item.result?.url) URL.revokeObjectURL(item.result.url);
   755→    }
   756→    void idbClearResults();
   757→    set({ items: [], compareId: null, totals: initialTotals });
   758→  },
   759→
   760→  retry: (id) => {
   761→    patchItem(id, {
   762→      status: 'queued',
   763→      progress: 0,
   764→      phase: 'Queued',
   765→      error: undefined,
   766→      result: undefined,
   767→      cpuRetry: false,
   768→      gpuRetry: false,
   769→      patch: null,
   770→    });
   771→    void ensureLoop();
   772→  },
   773→
   774→  toggleZip: (id) => {
   775→    useStore.setState((s) => ({
   776→      items: s.items.map((it) =>
   777→        it.id === id ? { ...it, zip: it.zip === false } : it
   778→      ),
   779→    }));
   780→  },
   781→
   782→  restorePersisted: async () => {
   783→    let rows: Array<PersistedResult & { id: string }> = [];
   784→    try {
   785→      rows = await idbLoadResults();
   786→    } catch {
   787→      return 0;
   788→    }
   789→    if (!rows.length) return 0;
   790→    const existing = new Set(useStore.getState().items.map((i) => i.id));
   791→    const items: QueueItem[] = [];
   792→    for (const row of rows) {
   793→      if (existing.has(row.id)) continue;
   794→      items.push({
   795→        id: row.id,
   796→        file: null,
   797→        name: row.name,
   798→        mime: row.mime,
   799→        sizeIn: row.sizeIn,
   800→        w: row.w,
   801→        h: row.h,
   802→        thumbUrl: URL.createObjectURL(row.thumb),
   803→        originalUrl: URL.createObjectURL(row.original),
   804→        status: 'done',
   805→        progress: 1,
   806→        phase: 'Done',
   807→        result: {
   808→          url: URL.createObjectURL(row.result.blob),
   809→          size: row.result.blob.size,
   810→          w: row.result.w,
   811→          h: row.result.h,
   812→          ms: row.result.ms,
   813→          scale: row.result.scale,
   814→          preset: row.result.preset,
   815→          format: row.result.format,
   816→          clamped: row.result.clamped,
   817→          target: row.result.target,
   818→        },
   819→        restored: true,
   820→      });
   821→    }
   822→    if (items.length) {
   823→      set((s) => ({ items: [...items, ...s.items] }));
   824→    }
   825→    return items.length;
   826→  },
   827→
   828→  setSettings: (patch) => {
   829→    set((s) => ({ settings: { ...s.settings, ...patch } }));
   830→  },
   831→
   832→  togglePause: () => {
   833→    const wasPaused = get().paused;
   834→    set({ paused: !wasPaused });
   835→    if (wasPaused) void ensureLoop();
   836→  },
   837→
   838→  setCompare: (id) => set({ compareId: id }),
   839→    }),
   840→    {
   841→      name: 'pixelforge-settings',
   842→      version: 1,
   843→      // persist only user preferences — never queue items (blob URLs die anyway)
   844→      partialize: (s) => ({ settings: s.settings }) as unknown as UpscalerState,
   845→      // hydration is triggered manually from <Workspace /> after mount so the
   846→      // server-rendered markup never mismatches
   847→      skipHydration: true,
   848→      merge: (persisted, current) => {
   849→        const p = (persisted ?? {}) as Partial<UpscalerState>;
   850→        return {
   851→          ...current,
   852→          ...p,
   853→          settings: { ...DEFAULT_SETTINGS, ...(p.settings ?? {}) },
   854→        };
   855→      },
   856→    }
   857→  )
   858→);

   863→export function downloadBlob(blob: Blob, filename: string) {
   864→  const url = URL.createObjectURL(blob);
   865→  const a = document.createElement('a');
   866→  a.href = url;
   867→  a.download = filename;
   868→  document.body.appendChild(a);
   869→  a.click();
   870→  a.remove();
   871→  setTimeout(() => URL.revokeObjectURL(url), 4000);
   872→}

   874→export function resultFilename(
   875→  name: string,
   876→  scale: ScaleFactor,
   877→  format: OutputFormat,
   878→  target?: number
   879→): string {
   880→  const base = name.replace(/\.[^.]+$/, '') || 'image';
   881→  const label = target ? `${target}px` : `${scale}x`;
   882→  return `${base}_${label}_upscaled.${extForFormat(format)}`;
   883→}

   885→export async function downloadAllAsZip() {
   886→  const { items } = useStore.getState();
   887→  const done = items.filter((i) => i.status === 'done' && i.result && i.zip !== false);
   888→  if (!done.length) return;
   889→  const { toast } = await import('sonner');
   890→  const t = toast.loading(`Packing ${done.length} image${done.length > 1 ? 's' : ''}…`);
   891→  try {
   892→    const JSZip = (await import('jszip')).default;
   893→    const zip = new JSZip();
   894→    const used = new Set<string>();
   895→    for (const item of done) {
   896→      const r = item.result!;
   897→      let filename = resultFilename(item.name, r.scale, r.format, r.target);
   898→      let i = 1;
   899→      while (used.has(filename)) {
   900→        filename = resultFilename(item.name, r.scale, r.format, r.target).replace(/(\.\w+)$/, `-${i++}$1`);
   901→      }
   902→      used.add(filename);
   903→      const blob = await (await fetch(r.url)).blob();
   904→      zip.file(filename, blob);
   905→    }
   906→    const out = await zip.generateAsync({ type: 'blob' });
   907→    const stamp = new Date().toISOString().slice(0, 16).replace(/[-:T]/g, '');
   908→    downloadBlob(out, `pixelforge-upscaled-${stamp}.zip`);
   909→    toast.success(`ZIP ready — ${formatBytes(out.size)}`, { id: t });
   910→  } catch (err) {
   911→    console.error(err);
   912→    toast.error('Could not build the ZIP archive', { id: t });
   913→  }
   914→}
