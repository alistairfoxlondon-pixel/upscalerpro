   147→   147→let workerWired = false;
   148→   148→const pending = new Map<string, { resolve: () => void }>();
   149→   149→const stallTimers = new Map<string, ReturnType<typeof setTimeout>>();
   150→   150→const STALL_TIMEOUT_MS = 300_000;
   151→   151→/** a fresh-context GPU retry after a Finalizing stall gets a shorter leash */
   152→   152→const GPU_RETRY_TIMEOUT_MS = 180_000;
   153→   153→let loopRunning = false;
   154→   154→
   155→   155→/* ---- patch-grid accumulation (flushed to the store at most ~8×/s) ---- */
   156→   156→const patchAccum = new Map<string, { cols: number; rows: number; done: number }>();
   157→   157→let patchFlushTimer: ReturnType<typeof setTimeout> | null = null;
   158→   158→
   159→   159→function flushPatches() {
   160→   160→  patchFlushTimer = null;
   161→   161→  if (!patchAccum.size) return;
   162→   162→  const entries = [...patchAccum.entries()];
   163→   163→  patchAccum.clear();
   164→   164→  useStore.setState((s) => ({
   165→   165→    items: s.items.map((it) => {
   166→   166→      const hit = entries.find(([id]) => id === it.id);
   167→   167→      return hit ? { ...it, patch: hit[1] } : it;
   168→   168→    }),
   169→   169→  }));
   170→   170→}
   171→   171→
   172→   172→function queuePatchFlush() {
   173→   173→  if (!patchFlushTimer) {
   174→   174→    patchFlushTimer = setTimeout(flushPatches, 120);
   175→   175→  }
   176→   176→}
   177→   177→
   178→   178→function onPatch(id: string, row: number, col: number, cols: number, rows: number) {
   179→   179→  if (!pending.has(id)) return;
   180→   180→  const prev = patchAccum.get(id);
   181→   181→  // a pass change (8× chain) swaps the grid dimensions — restart the count
   182→   182→  const base = prev && prev.cols === cols && prev.rows === rows ? prev : null;
   183→   183→  const done = Math.max(base?.done ?? 0, row * cols + col + 1);
   184→   184→  patchAccum.set(id, { cols, rows, done });
   185→   185→  queuePatchFlush();
   186→   186→}

   188→   188→function touchStall(id: string) {
   189→   189→  const prev = stallTimers.get(id);
   190→   190→  if (prev) clearTimeout(prev);
   191→   191→  const item = useStore.getState().items.find((i) => i.id === id);
   192→   192→  const timeout = item?.gpuRetry ? GPU_RETRY_TIMEOUT_MS : STALL_TIMEOUT_MS;
   193→   193→  stallTimers.set(
   194→   194→    id,
   195→   195→    setTimeout(() => {
   196→   196→      stallTimers.delete(id);
   197→   197→      onStall(id);
   198→   198→    }, timeout)
   199→   199→  );
   200→   200→}
   201→   201→
   202→   202→function clearStall(id: string) {
   203→   203→  const prev = stallTimers.get(id);
   204→   204→  if (prev) clearTimeout(prev);
   205→   205→  stallTimers.delete(id);
   206→   206→}
   207→   207→
   208→   208→async function onStall(id: string) {
   209→   209→  if (!pending.has(id)) return;
   210→   210→  const item = useStore.getState().items.find((i) => i.id === id);
   211→   211→  if (!item) return;
   212→   212→  const { toast } = await import('sonner');
   213→   213→  if (item.cpuRetry) {
   214→   214→    console.error('[pixelforge] processing stalled even on CPU fallback', id);
   215→   215→    patchItem(id, {
   216→   216→      status: 'error',
   217→   217→      phase: 'Failed',
   218→   218→      error: 'Processing stalled — try a smaller image or the Fast engine',
   219→   219→      result: undefined,
   220→   220→    });
   221→   221→    pending.get(id)?.resolve();
   222→   222→    pending.delete(id);
   223→   223→    return;
   224→   224→  }
   225→   225→  // The GPU readback is wedged — a cancel cannot unblock the worker's job
   226→   226→  // chain, so terminate it and give the retry a fresh worker.
   227→   227→  if (item.gpuRetry) {
   228→   228→    // Second stall — the fresh GPU context did not survive either. Fall back
   229→   229→    // to the pure-JS backend: slow but deterministic.
   230→   230→    toast.info('Graphics engine stalled again — retrying on CPU (slower but reliable)');
   231→   231→    resetWorker();
   232→   232→    patchItem(id, {
   233→   233→      cpuRetry: true,
   234→   234→      gpuRetry: false,
   235→   235→      status: 'queued',
   236→   236→      phase: 'Queued (CPU retry)',
   237→   237→      progress: 0,
   238→   238→      result: undefined,
   239→   239→    });
   240→   240→  } else if (/^Finalizing/i.test(item.phase)) {
   241→   241→    // The AI pass already finished — only the GPU→CPU readback is suspect.
   242→   242→    // A fresh WebGL context is usually healthy again, and re-running the AI
   243→   243→    // pass is far cheaper than a full CPU inference (which can take minutes).
   244→   244→    toast.info('Graphics engine stalled during output — retrying on a fresh GPU context');
   245→   245→    resetWorker();
   246→   246→    patchItem(id, {
   247→   247→      gpuRetry: true,
   248→   248→      status: 'queued',
   249→   249→      phase: 'Queued (GPU retry)',
   250→   250→      progress: 0,
   251→   251→      result: undefined,
   252→   252→    });
   253→   253→  } else {
   254→   254→    // Compute-phase stall — go straight to the reliable CPU fallback.
   255→   255→    toast.info('Graphics engine stalled — restarting worker and retrying on CPU (slower but reliable)');
   256→   256→    resetWorker();
   257→   257→    patchItem(id, {
   258→   258→      cpuRetry: true,
   259→   259→      gpuRetry: false,
   260→   260→      status: 'queued',
   261→   261→      phase: 'Queued (CPU retry)',
   262→   262→      progress: 0,
   263→   263→      result: undefined,
   264→   264→    });
   265→   265→  }
   266→   266→  pending.get(id)?.resolve();
   267→   267→  pending.delete(id);
   268→   268→}

   270→   270→function patchItem(id: string, patch: Partial<QueueItem>) {
   271→   271→  useStore.setState((s) => ({
   272→   272→    items: s.items.map((it) => (it.id === id ? { ...it, ...patch } : it)),
   273→   273→  }));
   274→   274→}

   276→   276→/**
   277→   277→ * Persist a finished result to IndexedDB so it can be restored after a
   278→   278→ * reload. The original is re-encoded to a capped preview; thumbnails are
   279→   279→ * already small. Never throws.
   280→   280→ */
   281→   281→async function savePersisted(item: QueueItem, blob: Blob, result: ResultData) {
   282→   282→  try {
   283→   283→    const [thumbBlob, originalBlob] = await Promise.all([
   284→   284→      fetch(item.thumbUrl).then((r) => r.blob()).catch(() => null),
   285→   285→      item.originalUrl
   286→   286→        ? makePreviewUrlFromUrl(item.originalUrl).catch(() => null)
   287→   287→        : Promise.resolve(null),
   288→   288→    ]);
   289→   289→    if (!thumbBlob || !originalBlob) return;
   290→   290→    const data: PersistedResult = {
   291→   291→      w: item.w,
   292→   292→      h: item.h,
   293→   293→      sizeIn: item.sizeIn,
   294→   294→      name: item.name,
   295→   295→      mime: item.mime,
   296→   296→      thumb: thumbBlob,
   297→   297→      original: originalBlob,
   298→   298→      result: {
   299→   299→        blob,
   300→   300→        w: result.w,
   301→   301→        h: result.h,
   302→   302→        ms: result.ms,
   303→   303→        scale: result.scale,
   304→   304→        preset: result.preset,
   305→   305→        format: result.format,
   306→   306→        clamped: result.clamped,
   307→   307→        target: result.target,
   308→   308→      },
   309→   309→      savedAt: Date.now(),
   310→   310→    };
   311→   311→    await idbPutResult(item.id, data);
   312→   312→  } catch {
   313→   313→    /* best-effort */
   314→   314→  }
   315→   315→}

   317→   317→/** Re-encodes any image object-URL to a ≤2048px JPEG for compact storage. */
   318→   318→async function makePreviewUrlFromUrl(url: string, maxSide = 2048): Promise<Blob> {
   319→   319→  const src = await createImageBitmap(await (await fetch(url)).blob());
   320→   320→  try {
   321→   321→    const ratio = Math.min(1, maxSide / Math.max(src.width, src.height));
   322→   322→    const w = Math.max(1, Math.round(src.width * ratio));
   323→   323→    const h = Math.max(1, Math.round(src.height * ratio));
   324→   324→    const canvas = document.createElement('canvas');
   325→   325→    canvas.width = w;
   326→   326→    canvas.height = h;
   327→   327→    const ctx = canvas.getContext('2d');
   328→   328→    if (!ctx) throw new Error('Canvas unavailable');
   329→   329→    ctx.imageSmoothingQuality = 'high';
   330→   330→    ctx.drawImage(src, 0, 0, w, h);
   331→   331→    const blob = await new Promise<Blob | null>((res) => canvas.toBlob(res, 'image/jpeg', 0.9));
   332→   332→    if (!blob) throw new Error('Preview encoding failed');
   333→   333→    return blob;
   334→   334→  } finally {
   335→   335→    src.close();
   336→   336→  }
   337→   337→}

   339→   339→function wireWorker() {
   340→   340→  if (workerWired) return;
   341→   341→  workerWired = true;
   342→   342→  onWorkerMessage((m: WorkerOutMessage) => {
   343→   343→    const state = useStore.getState();
   344→   344→    switch (m.type) {
   345→   345→      case 'backend':
   346→   346→        useStore.setState({ backend: m.backend, gpu: m.renderer ?? null });
   347→   347→        break;
   348→   348→      case 'debug':
   349→   349→        console.info('[pf-worker]', m.text);
   350→   350→        break;
   351→   351→      case 'phase':
   352→   352→        if (pending.has(m.id)) {
   353→   353→          touchStall(m.id);
   354→   354→          patchItem(m.id, { phase: m.phase });
   355→   355→        }
   356→   356→        break;
   357→   357→      case 'model-status': {
   358→   358→        const key = `${m.preset}:${m.scale}`;
   359→   359→        if (state.modelKey === key || m.status === 'error') {
   360→   360→          useStore.setState({ modelStatus: m.status, modelMessage: m.message ?? null });
   361→   361→        }
   362→   362→        if (m.status === 'loading' && state.activeId) {
   363→   363→          patchItem(state.activeId, { phase: 'Loading AI model' });
   364→   364→          touchStall(state.activeId);
   365→   365→        }
   366→   366→        if (m.status === 'ready' && state.activeId) {
   367→   367→          const active = state.items.find((i) => i.id === state.activeId);
   368→   368→          if (active && active.phase === 'Loading AI model') {
   369→   369→            patchItem(active.id, { phase: 'Upscaling' });
   370→   370→          }
   371→   371→        }
   372→   372→        break;
   373→   373→      }
   374→   374→      case 'progress':
   375→   375→        touchStall(m.id);
   376→   376→        patchItem(m.id, {
   377→   377→          progress: 0.02 + m.rate * 0.93,
   378→   378→          phase: m.rate >= 0.999 ? 'Finalizing' : 'Upscaling',
   379→   379→        });
   380→   380→        break;
   381→   381→      case 'patch':
   382→   382→        onPatch(m.id, m.row, m.col, m.cols, m.rows);
   383→   383→        break;
   384→   384→      case 'done': {
   385→   385→        if (!pending.has(m.id)) break; // stale message (item re-queued or removed)
   386→   386→        clearStall(m.id);
   387→   387→        patchAccum.delete(m.id);
   388→   388→        const item = state.items.find((i) => i.id === m.id);
   389→   389→        const url = URL.createObjectURL(m.blob);
   390→   390→        const planned = item?.result;
   391→   391→        const result: ResultData = {
   392→   392→          url,
   393→   393→          size: m.blob.size,
   394→   394→          w: m.width,
   395→   395→          h: m.height,
   396→   396→          ms: m.ms,
   397→   397→          scale: (planned?.scale ?? useStore.getState().settings.scale) as ScaleFactor,
   398→   398→          preset: (planned?.preset ?? useStore.getState().settings.preset) as PresetId,
   399→   399→          format: m.blob.type.includes('jpeg')
   400→   400→            ? 'jpeg'
   401→   401→            : m.blob.type.includes('webp')
   402→   402→              ? 'webp'
   403→   403→              : 'png',
   404→   404→          clamped: planned?.clamped ?? false,
   405→   405→          target: planned?.target,
   406→   406→        };
   407→   407→        patchItem(m.id, {
   408→   408→          status: 'done',
   409→   409→          progress: 1,
   410→   410→          phase: 'Done',
   411→   411→          result,
   412→   412→          patch: null,
   413→   413→        });
   414→   414→        useStore.setState((s) => ({
   415→   415→          totals: {
   416→   416→            images: s.totals.images + 1,
   417→   417→            pixelsIn: s.totals.pixelsIn + (item ? item.w * item.h : 0),
   418→   418→            pixelsOut: s.totals.pixelsOut + m.width * m.height,
   419→   419→            ms: s.totals.ms + m.ms,
   420→   420→            bytesIn: s.totals.bytesIn + (item?.sizeIn ?? 0),
   421→   421→            bytesOut: s.totals.bytesOut + m.blob.size,
   422→   422→          },
   423→   423→        }));
   424→   424→        pending.get(m.id)?.resolve();
   425→   425→        pending.delete(m.id);
   426→   426→        // persist the finished result so it survives a reload (best-effort)
   427→   427→        if (item) void savePersisted(item, m.blob, result);
   428→   428→        break;
   429→   429→      }
   430→   430→      case 'error': {
   431→   431→        if (!pending.has(m.id)) break; // stale message (item re-queued or removed)
   432→   432→        clearStall(m.id);
   433→   433→        patchAccum.delete(m.id);
   434→   434→        const canceled = /^canceled$/i.test(m.message);
   435→   435→        patchItem(m.id, {
   436→   436→          status: canceled ? 'canceled' : 'error',
   437→   437→          phase: canceled ? 'Canceled' : 'Failed',
   438→   438→          error: canceled ? undefined : m.message,
   439→   439→          progress: 0,
   440→   440→          result: undefined,
   441→   441→        });
   442→   442→        if (!canceled) {
   443→   443→          console.error('[pixelforge]', m.message);
   444→   444→          void import('sonner').then(({ toast }) =>
   445→   445→            toast.error('Upscaling failed', { description: m.message })
   446→   446→          );
   447→   447→        }
   448→   448→        pending.get(m.id)?.resolve();
   449→   449→        pending.delete(m.id);
   450→   450→        break;
   451→   451→      }
   452→   452→    }
   453→   453→  });
   454→   454→}

   456→   456→async function processItem(id: string): Promise<void> {
   457→   457→  const state = useStore.getState();
   458→   458→  const item = state.items.find((i) => i.id === id);
   459→   459→  if (!item) return;
   460→   460→  if (!item.file) {
   461→   461→    // restored items have no source file — treat as informational failure
   462→   462→    patchItem(id, {
   463→   463→      status: 'error',
   464→   464→      phase: 'Failed',
   465→   465→      error: 'Original file is not kept after a reload — add the image again',
   466→   466→    });
   467→   467→    return;
   468→   468→  }
   469→   469→  const { preset, scale, format, jpegQuality, denoise, sharpen, keepExif } = state.settings;
   470→   470→
   471→   471→  patchItem(id, {
   472→   472→    status: 'processing',
   473→   473→    progress: 0,
   474→   474→    phase: 'Preparing',
   475→   475→    error: undefined,
   476→   476→    patch: null,
   477→   477→  });
   478→   478→  useStore.setState({ activeId: id });
   479→   479→
   480→   480→  const resolved: OutputFormat = format === 'auto' ? resolveFormat(item.mime, item.name) : format;
   481→   481→
   482→   482→  // memory safety: pick the largest allowed scale (factor mode) or the smallest
   483→   483→  // AI scale that meets the target size, then resize (target mode)
   484→   484→  const fit: SizeFit | null =
   485→   485→    state.settings.scaleMode === 'target'
   486→   486→      ? resolveScaleForTarget(item.w, item.h, state.settings.targetSide)
   487→   487→      : resolveScale(item.w, item.h, scale);
   488→   488→  if (!fit || fit.error) {
   489→   489→    patchItem(id, {
   490→   490→      status: 'error',
   491→   491→      phase: 'Failed',
   492→   492→      error: fit?.error ?? 'Image is too large to upscale on this device',
   493→   493→    });
   494→   494→    return;
   495→   495→  }
   496→   496→
   497→   497→  try {
   498→   498→    const { bitmap } = await decodeImageFile(item.file);
   499→   499→    const modelKey = `${preset}:${fit.scale}`;
   500→   500→    useStore.setState({ modelKey, modelStatus: null, modelMessage: null });
   501→   501→    // planned result placeholder — carries the plan (scale/format) and
   502→   502→    // estimated output size until the worker reports the real values
   503→   503→    const planned: ResultData = {
   504→   504→      url: '',
   505→   505→      size: 0,
   506→   506→      w: fit.outW,
   507→   507→      h: fit.outH,
   508→   508→      ms: 0,
   509→   509→      scale: fit.scale,
   510→   510→      preset,
   511→   511→      format: resolved,
   512→   512→      clamped: fit.clamped ?? false,
   513→   513→      target:
   514→   514→        state.settings.scaleMode === 'target' && !fit.short && !fit.error
   515→   515→          ? state.settings.targetSide
   516→   516→          : undefined,
   517→   517→    };
   518→   518→    patchItem(id, {
   519→   519→      phase: 'Loading model',
   520→   520→      progress: 0.01,
   521→   521→      result: planned,
   522→   522→    });
   523→   523→    // Opt-in metadata preservation: pull the original JPEG APP1 EXIF segment
   524→   524→    // (first 256 KB header scan — cheap) so the worker can splice it into the
   525→   525→    // encoded output. Only meaningful for JPEG results.
   526→   526→    const exif =
   527→   527→      keepExif && resolved === 'jpeg' ? await extractExifSegment(item.file) : null;
   528→   528→    const done = new Promise<void>((resolve) => pending.set(id, { resolve }));
   529→   529→    touchStall(id);
   530→   530→    const transfer: Transferable[] = [bitmap];
   531→   531→    if (exif) transfer.push(exif);
   532→   532→    postToWorker(
   533→   533→      {
   534→   534→        type: 'upscale',
   535→   535→        id,
   536→   536→        bitmap,
   537→   537→        scale: fit.scale,
   538→   538→        preset,
   539→   539→        format: resolved,
   540→   540→        quality: jpegQuality,
   541→   541→        denoise,
   542→   542→        sharpen,
   543→   543→        exif: exif ?? undefined,
   544→   544→        targetSide: planned.target,
   545→   545→        backendHint: item.cpuRetry ? 'cpu' : undefined,
   546→   546→      },
   547→   547→      transfer
   548→   548→    );
   549→   549→    await done;
   550→   550→    clearStall(id);
   551→   551→  } catch (err) {
   552→   552→    patchItem(id, {
   553→   553→      status: 'error',
   554→   554→      phase: 'Failed',
   555→   555→      error: err instanceof Error ? err.message : 'Could not decode image',
   556→   556→      result: undefined,
   557→   557→    });
   558→   558→  }
   559→   559→}

   561→   561→async function ensureLoop() {
   562→   562→  if (loopRunning) return;
   563→   563→  loopRunning = true;
   564→   564→  useStore.setState({ busy: true });
   565→   565→  let completed = 0;
   566→   566→  try {
   567→   567→    for (;;) {
   568→   568→      const s = useStore.getState();
   569→   569→      if (s.paused) break;
   570→   570→      const next = s.items.find((i) => i.status === 'queued');
   571→   571→      if (!next) break;
   572→   572→      await processItem(next.id);
   573→   573→      const after = useStore.getState().items.find((i) => i.id === next.id);
   574→   574→      if (after?.status === 'done') completed++;
   575→   575→    }
   576→   576→  } finally {
   577→   577→    loopRunning = false;
   578→   578→    useStore.setState({ busy: false, activeId: null });
   579→   579→    // batch-complete feedback with a one-tap ZIP export
   580→   580→    if (completed > 0 && !useStore.getState().paused) {
   581→   581→      const { toast } = await import('sonner');
   582→   582→      toast.success(completed === 1 ? '1 image ready' : `${completed} images ready`, {
   583→   583→        description: 'Processed entirely on this device.',
   584→   584→        action: { label: 'Download ZIP', onClick: () => void downloadAllAsZip() },
   585→   585→        duration: 10_000,
   586→   586→      });
   587→   587→    }
   588→   588→  }
   589→   589→}

   610→   610→  backend: null,

   612→   612→  modelKey: null,
   613→   613→  modelStatus: null,
   614→   614→  modelMessage: null,
   615→   615→  totals: initialTotals,
   616→   616→  compareId: null,
   617→   617→
   618→   618→  addFiles: async (files) => {
   619→   619→    wireWorker();
   620→   620→    const list = files.filter(
   621→   621→      (f) =>
   622→   622→        f.size > 0 &&
   623→   623→        (f.type.startsWith('image/') || /\.(heic|heif|hif|tif|tiff|avif|webp|bmp)$/i.test(f.name))
   624→   624→    );

   626→   626→      const { toast } = await import('sonner');

   628→   628→        description: 'Drop JPEG, PNG, WebP, AVIF, GIF, BMP, HEIC or TIFF files.',
   629→   629→      });

   631→   631→    }

   633→   633→    // duplicate guard: same name + byte size already queued (or twice in this
   634→   634→    // batch) — a re-added file would just waste GPU time on an identical job
   635→   635→    const seen = new Set<string>();

   637→   637→      if (it.file) seen.add(`${it.name}:${it.sizeIn}`);
   638→   638→    }

   642→   642→      const key = `${file.name || 'image'}:${file.size}`;
   643→   643→      if (seen.has(key)) {
   644→   644→        dupes++;
   645→   645→        continue;
   646→   646→      }
   647→   647→      seen.add(key);
   648→   648→      const id = crypto.randomUUID();
   649→   649→      const name = file.name || `image-${id.slice(0, 6)}`;
   650→   650→      try {
   651→   651→        const { bitmap, renderable } = await decodeImageFile(file);
   652→   652→        const pixels = bitmap.width * bitmap.height;
   653→   653→        if (pixels > MAX_INPUT_PIXELS) {
   654→   654→          bitmap.close();
   655→   655→          items.push({
   656→   656→            id, file, name, mime: file.type || 'image/*', sizeIn: file.size,
   657→   657→            w: 0, h: 0, thumbUrl: '', originalUrl: '',
   658→   658→            status: 'error', progress: 0, phase: 'Failed',
   659→   659→            error: `Image is ${Math.round(pixels / 1e6)}MP — max input is ${Math.round(MAX_INPUT_PIXELS / 1e6)}MP`,
   660→   660→          });
   661→   661→          continue;
   662→   662→        }
   663→   663→        const thumbUrl = await makeThumbUrl(bitmap);
   664→   664→        const originalUrl = renderable ? URL.createObjectURL(file) : await makePreviewUrl(bitmap);
   665→   665→        const w = bitmap.width;
   666→   666→        const h = bitmap.height;
   667→   667→        bitmap.close();
   668→   668→        items.push({
   669→   669→          id, file, name, mime: file.type || 'image/*', sizeIn: file.size,
   670→   670→          w, h, thumbUrl, originalUrl,
   671→   671→          status: 'queued', progress: 0, phase: 'Queued',
   672→   672→        });
   673→   673→      } catch (err) {
   674→   674→        items.push({
   675→   675→          id, file, name, mime: file.type || 'image/*', sizeIn: file.size,
   676→   676→          w: 0, h: 0, thumbUrl: '', originalUrl: '',
   677→   677→          status: 'error', progress: 0, phase: 'Failed',
   678→   678→          error: err instanceof Error ? err.message : 'Could not decode image',
   679→   679→        });
   680→   680→      }
   681→   681→    }
   682→   682→    set((s) => ({ items: [...s.items, ...items] }));

   686→   686→        description: 'Already in the queue.',
   687→   687→      });

   690→   690→    if (skipped > 0) {
   691→   691→      const { toast } = await import('sonner');

   693→   693→        description: `Batch limit is ${MAX_BATCH_FILES} images at a time.`,
   694→   694→      });
   695→   695→    }
   696→   696→    void ensureLoop();
   697→   697→  },

   863→   863→export function downloadBlob(blob: Blob, filename: string) {
   864→   864→  const url = URL.createObjectURL(blob);
   865→   865→  const a = document.createElement('a');
   866→   866→  a.href = url;
   867→   867→  a.download = filename;
   868→   868→  document.body.appendChild(a);
   869→   869→  a.click();
   870→   870→  a.remove();
   871→   871→  setTimeout(() => URL.revokeObjectURL(url), 4000);
   872→   872→}

   874→   874→export function resultFilename(
   875→   875→  name: string,
   876→   876→  scale: ScaleFactor,
   877→   877→  format: OutputFormat,
   878→   878→  target?: number
   879→   879→): string {
   880→   880→  const base = name.replace(/\.[^.]+$/, '') || 'image';
   881→   881→  const label = target ? `${target}px` : `${scale}x`;
   882→   882→  return `${base}_${label}_upscaled.${extForFormat(format)}`;
   883→   883→}

   885→   885→export async function downloadAllAsZip() {
   886→   886→  const { items } = useStore.getState();
   887→   887→  const done = items.filter((i) => i.status === 'done' && i.result && i.zip !== false);
   888→   888→  if (!done.length) return;
   889→   889→  const { toast } = await import('sonner');
   890→   890→  const t = toast.loading(`Packing ${done.length} image${done.length > 1 ? 's' : ''}…`);
   891→   891→  try {
   892→   892→    const JSZip = (await import('jszip')).default;
   893→   893→    const zip = new JSZip();
   894→   894→    const used = new Set<string>();
   895→   895→    for (const item of done) {
   896→   896→      const r = item.result!;
   897→   897→      let filename = resultFilename(item.name, r.scale, r.format, r.target);
   898→   898→      let i = 1;
   899→   899→      while (used.has(filename)) {
   900→   900→        filename = resultFilename(item.name, r.scale, r.format, r.target).replace(/(\.\w+)$/, `-${i++}$1`);
   901→   901→      }
   902→   902→      used.add(filename);
   903→   903→      const blob = await (await fetch(r.url)).blob();
   904→   904→      zip.file(filename, blob);
   905→   905→    }
   906→   906→    const out = await zip.generateAsync({ type: 'blob' });
   907→   907→    const stamp = new Date().toISOString().slice(0, 16).replace(/[-:T]/g, '');
   908→   908→    downloadBlob(out, `pixelforge-upscaled-${stamp}.zip`);
   909→   909→    toast.success(`ZIP ready — ${formatBytes(out.size)}`, { id: t });
   910→   910→  } catch (err) {
   911→   911→    console.error(err);
   912→   912→    toast.error('Could not build the ZIP archive', { id: t });
   913→   913→  }
   914→   914→}
