   130→let workerWired = false;
   131→const pending = new Map<string, { resolve: () => void }>();
   132→const stallTimers = new Map<string, ReturnType<typeof setTimeout>>();
   133→const STALL_TIMEOUT_MS = 300_000;
   134→let loopRunning = false;
   135→
   136→/* ---- patch-grid accumulation (flushed to the store at most ~8×/s) ---- */
   137→const patchAccum = new Map<string, { cols: number; rows: number; done: number }>();
   138→let patchFlushTimer: ReturnType<typeof setTimeout> | null = null;
   139→
   140→function flushPatches() {
   141→  patchFlushTimer = null;
   142→  if (!patchAccum.size) return;
   143→  const entries = [...patchAccum.entries()];
   144→  patchAccum.clear();
   145→  useStore.setState((s) => ({
   146→    items: s.items.map((it) => {
   147→      const hit = entries.find(([id]) => id === it.id);
   148→      return hit ? { ...it, patch: hit[1] } : it;
   149→    }),
   150→  }));
   151→}
   152→
   153→function queuePatchFlush() {
   154→  if (!patchFlushTimer) {
   155→    patchFlushTimer = setTimeout(flushPatches, 120);
   156→  }
   157→}
   158→
   159→function onPatch(id: string, row: number, col: number, cols: number, rows: number) {
   160→  if (!pending.has(id)) return;
   161→  const prev = patchAccum.get(id);
   162→  // a pass change (8× chain) swaps the grid dimensions — restart the count
   163→  const base = prev && prev.cols === cols && prev.rows === rows ? prev : null;
   164→  const done = Math.max(base?.done ?? 0, row * cols + col + 1);
   165→  patchAccum.set(id, { cols, rows, done });
   166→  queuePatchFlush();
   167→}
   168→
   169→function touchStall(id: string) {
   170→  const prev = stallTimers.get(id);
   171→  if (prev) clearTimeout(prev);
   172→  stallTimers.set(
   173→    id,
   174→    setTimeout(() => {
   175→      stallTimers.delete(id);
   176→      onStall(id);
   177→    }, STALL_TIMEOUT_MS)
   178→  );
   179→}
   180→
   181→function clearStall(id: string) {
   182→  const prev = stallTimers.get(id);
   183→  if (prev) clearTimeout(prev);
   184→  stallTimers.delete(id);
   185→}
   186→
   187→async function onStall(id: string) {
   188→  if (!pending.has(id)) return;
   189→  const item = useStore.getState().items.find((i) => i.id === id);
   190→  if (!item) return;
   191→  const { toast } = await import('sonner');
   192→  if (item.cpuRetry) {
   193→    console.error('[pixelforge] processing stalled even on CPU fallback', id);
   194→    patchItem(id, {
   195→      status: 'error',
   196→      phase: 'Failed',
   197→      error: 'Processing stalled — try a smaller image or the Fast engine',
   198→      result: undefined,
   199→    });
   200→    pending.get(id)?.resolve();
   201→    pending.delete(id);
   202→    return;
   203→  }
   204→  // The GPU readback is wedged — a cancel cannot unblock the worker's job
   205→  // chain, so terminate it and give the retry a fresh worker + CPU backend.
   206→  toast.info('Graphics engine stalled — restarting worker and retrying on CPU (slower but reliable)');
   207→  resetWorker();
   208→  patchItem(id, {
   209→    cpuRetry: true,
   210→    status: 'queued',
   211→    phase: 'Queued (CPU retry)',
   212→    progress: 0,
   213→    result: undefined,
   214→  });
   215→  pending.get(id)?.resolve();
   216→  pending.delete(id);
   217→}
   218→
   219→function patchItem(id: string, patch: Partial<QueueItem>) {
   220→  useStore.setState((s) => ({
   221→    items: s.items.map((it) => (it.id === id ? { ...it, ...patch } : it)),
   222→  }));
   223→}

   225→/**
   226→ * Persist a finished result to IndexedDB so it can be restored after a
   227→ * reload. The original is re-encoded to a capped preview; thumbnails are
   228→ * already small. Never throws.
   229→ */
   230→async function savePersisted(item: QueueItem, blob: Blob, result: ResultData) {
   231→  try {
   232→    const [thumbBlob, originalBlob] = await Promise.all([
   233→      fetch(item.thumbUrl).then((r) => r.blob()).catch(() => null),
   234→      item.originalUrl
   235→        ? makePreviewUrlFromUrl(item.originalUrl).catch(() => null)
   236→        : Promise.resolve(null),
   237→    ]);
   238→    if (!thumbBlob || !originalBlob) return;
   239→    const data: PersistedResult = {
   240→      w: item.w,
   241→      h: item.h,
   242→      sizeIn: item.sizeIn,
   243→      name: item.name,
   244→      mime: item.mime,
   245→      thumb: thumbBlob,
   246→      original: originalBlob,
   247→      result: {
   248→        blob,
   249→        w: result.w,
   250→        h: result.h,
   251→        ms: result.ms,
   252→        scale: result.scale,
   253→        preset: result.preset,
   254→        format: result.format,
   255→        clamped: result.clamped,
   256→      },
   257→      savedAt: Date.now(),
   258→    };
   259→    await idbPutResult(item.id, data);
   260→  } catch {
   261→    /* best-effort */
   262→  }
   263→}

   265→/** Re-encodes any image object-URL to a ≤2048px JPEG for compact storage. */
   266→async function makePreviewUrlFromUrl(url: string, maxSide = 2048): Promise<Blob> {
   267→  const src = await createImageBitmap(await (await fetch(url)).blob());
   268→  try {
   269→    const ratio = Math.min(1, maxSide / Math.max(src.width, src.height));
   270→    const w = Math.max(1, Math.round(src.width * ratio));
   271→    const h = Math.max(1, Math.round(src.height * ratio));
   272→    const canvas = document.createElement('canvas');
   273→    canvas.width = w;
   274→    canvas.height = h;
   275→    const ctx = canvas.getContext('2d');
   276→    if (!ctx) throw new Error('Canvas unavailable');
   277→    ctx.imageSmoothingQuality = 'high';
   278→    ctx.drawImage(src, 0, 0, w, h);
   279→    const blob = await new Promise<Blob | null>((res) => canvas.toBlob(res, 'image/jpeg', 0.9));
   280→    if (!blob) throw new Error('Preview encoding failed');
   281→    return blob;
   282→  } finally {
   283→    src.close();
   284→  }
   285→}

   287→function wireWorker() {
   288→  if (workerWired) return;
   289→  workerWired = true;
   290→  onWorkerMessage((m: WorkerOutMessage) => {
   291→    const state = useStore.getState();
   292→    switch (m.type) {
   293→      case 'backend':
   294→        useStore.setState({ backend: m.backend, gpu: m.renderer ?? null });
   295→        break;
   296→      case 'debug':
   297→        console.info('[pf-worker]', m.text);
   298→        break;
   299→      case 'phase':
   300→        if (pending.has(m.id)) {
   301→          touchStall(m.id);
   302→          patchItem(m.id, { phase: m.phase });
   303→        }
   304→        break;
   305→      case 'model-status': {
   306→        const key = `${m.preset}:${m.scale}`;
   307→        if (state.modelKey === key || m.status === 'error') {
   308→          useStore.setState({ modelStatus: m.status, modelMessage: m.message ?? null });
   309→        }
   310→        if (m.status === 'loading' && state.activeId) {
   311→          patchItem(state.activeId, { phase: 'Loading AI model' });
   312→          touchStall(state.activeId);
   313→        }
   314→        if (m.status === 'ready' && state.activeId) {
   315→          const active = state.items.find((i) => i.id === state.activeId);
   316→          if (active && active.phase === 'Loading AI model') {
   317→            patchItem(active.id, { phase: 'Upscaling' });
   318→          }
   319→        }
   320→        break;
   321→      }
   322→      case 'progress':
   323→        touchStall(m.id);
   324→        patchItem(m.id, {
   325→          progress: 0.02 + m.rate * 0.93,
   326→          phase: m.rate >= 0.999 ? 'Finalizing' : 'Upscaling',
   327→        });
   328→        break;
   329→      case 'patch':
   330→        onPatch(m.id, m.row, m.col, m.cols, m.rows);
   331→        break;
   332→      case 'done': {
   333→        if (!pending.has(m.id)) break; // stale message (item re-queued or removed)
   334→        clearStall(m.id);
   335→        patchAccum.delete(m.id);
   336→        const item = state.items.find((i) => i.id === m.id);
   337→        const url = URL.createObjectURL(m.blob);
   338→        const planned = item?.result;
   339→        const result: ResultData = {
   340→          url,
   341→          size: m.blob.size,
   342→          w: m.width,
   343→          h: m.height,
   344→          ms: m.ms,
   345→          scale: (planned?.scale ?? useStore.getState().settings.scale) as ScaleFactor,
   346→          preset: (planned?.preset ?? useStore.getState().settings.preset) as PresetId,
   347→          format: m.blob.type.includes('jpeg')
   348→            ? 'jpeg'
   349→            : m.blob.type.includes('webp')
   350→              ? 'webp'
   351→              : 'png',
   352→          clamped: planned?.clamped ?? false,
   353→        };
   354→        patchItem(m.id, {
   355→          status: 'done',
   356→          progress: 1,
   357→          phase: 'Done',
   358→          result,
   359→          patch: null,
   360→        });
   361→        useStore.setState((s) => ({
   362→          totals: {
   363→            images: s.totals.images + 1,
   364→            pixelsIn: s.totals.pixelsIn + (item ? item.w * item.h : 0),
   365→            pixelsOut: s.totals.pixelsOut + m.width * m.height,
   366→            ms: s.totals.ms + m.ms,
   367→            bytesIn: s.totals.bytesIn + (item?.sizeIn ?? 0),
   368→            bytesOut: s.totals.bytesOut + m.blob.size,
   369→          },
   370→        }));
   371→        pending.get(m.id)?.resolve();
   372→        pending.delete(m.id);
   373→        // persist the finished result so it survives a reload (best-effort)
   374→        if (item) void savePersisted(item, m.blob, result);
   375→        break;
   376→      }
   377→      case 'error': {
   378→        if (!pending.has(m.id)) break; // stale message (item re-queued or removed)
   379→        clearStall(m.id);
   380→        patchAccum.delete(m.id);
   381→        const canceled = /^canceled$/i.test(m.message);
   382→        patchItem(m.id, {
   383→          status: canceled ? 'canceled' : 'error',
   384→          phase: canceled ? 'Canceled' : 'Failed',
   385→          error: canceled ? undefined : m.message,
   386→          progress: 0,
   387→          result: undefined,
   388→        });
   389→        if (!canceled) {
   390→          console.error('[pixelforge]', m.message);
   391→          void import('sonner').then(({ toast }) =>
   392→            toast.error('Upscaling failed', { description: m.message })
   393→          );
   394→        }
   395→        pending.get(m.id)?.resolve();
   396→        pending.delete(m.id);
   397→        break;
   398→      }
   399→    }
   400→  });
   401→}

   403→async function processItem(id: string): Promise<void> {
   404→  const state = useStore.getState();
   405→  const item = state.items.find((i) => i.id === id);
   406→  if (!item) return;
   407→  if (!item.file) {
   408→    // restored items have no source file — treat as informational failure
   409→    patchItem(id, {
   410→      status: 'error',
   411→      phase: 'Failed',
   412→      error: 'Original file is not kept after a reload — add the image again',
   413→    });
   414→    return;
   415→  }
   416→  const { preset, scale, format, jpegQuality, denoise, sharpen } = state.settings;
   417→
   418→  patchItem(id, {
   419→    status: 'processing',
   420→    progress: 0,
   421→    phase: 'Preparing',
   422→    error: undefined,
   423→    patch: null,
   424→  });
   425→  useStore.setState({ activeId: id });
   426→
   427→  const resolved: OutputFormat = format === 'auto' ? resolveFormat(item.mime, item.name) : format;
   428→
   429→  // memory safety: pick the largest allowed scale
   430→  const fit = resolveScale(item.w, item.h, scale);
   431→  if (!fit) {
   432→    patchItem(id, {
   433→      status: 'error',
   434→      phase: 'Failed',
   435→      error: 'Image is too large to upscale on this device',
   436→    });
   437→    return;
   438→  }
   439→
   440→  try {
   441→    const { bitmap } = await decodeImageFile(item.file);
   442→    const modelKey = `${preset}:${fit.scale}`;
   443→    useStore.setState({ modelKey, modelStatus: null, modelMessage: null });
   444→    patchItem(id, {
   445→      phase: 'Loading model',
   446→      progress: 0.01,
   447→      // planned result placeholder — carries the plan (scale/format) and
   448→      // estimated output size until the worker reports the real values
   449→      result: {
   450→        url: '',
   451→        size: 0,
   452→        w: item.w * fit.scale,
   453→        h: item.h * fit.scale,
   454→        ms: 0,
   455→        scale: fit.scale,
   456→        preset,
   457→        format: resolved,
   458→        clamped: fit.clamped,
   459→      },
   460→    });
   461→    const done = new Promise<void>((resolve) => pending.set(id, { resolve }));
   462→    touchStall(id);
   463→    postToWorker(
   464→      {
   465→        type: 'upscale',
   466→        id,
   467→        bitmap,
   468→        scale: fit.scale,
   469→        preset,
   470→        format: resolved,
   471→        quality: jpegQuality,
   472→        denoise,
   473→        sharpen,
   474→        backendHint: item.cpuRetry ? 'cpu' : undefined,
   475→      },
   476→      [bitmap]
   477→    );
   478→    await done;
   479→    clearStall(id);
   480→  } catch (err) {
   481→    patchItem(id, {
   482→      status: 'error',
   483→      phase: 'Failed',
   484→      error: err instanceof Error ? err.message : 'Could not decode image',
   485→      result: undefined,
   486→    });
   487→  }
   488→}

   490→async function ensureLoop() {
   491→  if (loopRunning) return;
   492→  loopRunning = true;
   493→  useStore.setState({ busy: true });
   494→  let completed = 0;
   495→  try {
   496→    for (;;) {
   497→      const s = useStore.getState();
   498→      if (s.paused) break;
   499→      const next = s.items.find((i) => i.status === 'queued');
   500→      if (!next) break;
   501→      await processItem(next.id);
   502→      const after = useStore.getState().items.find((i) => i.id === next.id);
   503→      if (after?.status === 'done') completed++;
   504→    }
   505→  } finally {
   506→    loopRunning = false;
   507→    useStore.setState({ busy: false, activeId: null });
   508→    // batch-complete feedback with a one-tap ZIP export
   509→    if (completed > 0 && !useStore.getState().paused) {
   510→      const { toast } = await import('sonner');
   511→      toast.success(completed === 1 ? '1 image ready' : `${completed} images ready`, {
   512→        description: 'Processed entirely on this device.',
   513→        action: { label: 'Download ZIP', onClick: () => void downloadAllAsZip() },
   514→        duration: 10_000,
   515→      });
   516→    }
   517→  }
   518→}

   531→export const useStore = create<UpscalerState>()(
   532→  persist(
   533→    (set, get) => ({
   534→  items: [],
   535→  settings: { ...DEFAULT_SETTINGS },
   536→  busy: false,
   537→  paused: false,
   538→  activeId: null,
   539→  backend: null,
   540→  gpu: null,
   541→  modelKey: null,
   542→  modelStatus: null,
   543→  modelMessage: null,
   544→  totals: initialTotals,
   545→  compareId: null,
   546→
   547→  addFiles: async (files) => {
   548→    wireWorker();
   549→    const list = files.filter(
   550→      (f) =>
   551→        f.size > 0 &&
   552→        (f.type.startsWith('image/') || /\.(heic|heif|hif|tif|tiff|avif|webp|bmp)$/i.test(f.name))
   553→    );
   554→    if (!list.length) {
   555→      const { toast } = await import('sonner');
   556→      toast.error('No supported images found', {
   557→        description: 'Drop JPEG, PNG, WebP, AVIF, GIF, BMP, HEIC or TIFF files.',
   558→      });
   559→      return;
   560→    }
   561→    const capped = list.slice(0, MAX_BATCH_FILES);
   562→    const items: QueueItem[] = [];
   563→    for (const file of capped) {
   564→      const id = crypto.randomUUID();
   565→      const name = file.name || `image-${id.slice(0, 6)}`;
   566→      try {
   567→        const { bitmap, renderable } = await decodeImageFile(file);
   568→        const pixels = bitmap.width * bitmap.height;
   569→        if (pixels > MAX_INPUT_PIXELS) {
   570→          bitmap.close();
   571→          items.push({
   572→            id, file, name, mime: file.type || 'image/*', sizeIn: file.size,
   573→            w: 0, h: 0, thumbUrl: '', originalUrl: '',
   574→            status: 'error', progress: 0, phase: 'Failed',
   575→            error: `Image is ${Math.round(pixels / 1e6)}MP — max input is ${Math.round(MAX_INPUT_PIXELS / 1e6)}MP`,
   576→          });
   577→          continue;
   578→        }
   579→        const thumbUrl = await makeThumbUrl(bitmap);
   580→        const originalUrl = renderable ? URL.createObjectURL(file) : await makePreviewUrl(bitmap);
   581→        const w = bitmap.width;
   582→        const h = bitmap.height;
   583→        bitmap.close();
   584→        items.push({
   585→          id, file, name, mime: file.type || 'image/*', sizeIn: file.size,
   586→          w, h, thumbUrl, originalUrl,
   587→          status: 'queued', progress: 0, phase: 'Queued',
   588→        });
   589→      } catch (err) {
   590→        items.push({
   591→          id, file, name, mime: file.type || 'image/*', sizeIn: file.size,
   592→          w: 0, h: 0, thumbUrl: '', originalUrl: '',
   593→          status: 'error', progress: 0, phase: 'Failed',
   594→          error: err instanceof Error ? err.message : 'Could not decode image',
   595→        });
   596→      }
   597→    }

   771→export function downloadBlob(blob: Blob, filename: string) {
   772→  const url = URL.createObjectURL(blob);
   773→  const a = document.createElement('a');
   774→  a.href = url;
   775→  a.download = filename;
   776→  document.body.appendChild(a);
   777→  a.click();
   778→  a.remove();
   779→  setTimeout(() => URL.revokeObjectURL(url), 4000);
   780→}

   787→export async function downloadAllAsZip() {
   788→  const { items } = useStore.getState();
   789→  const done = items.filter((i) => i.status === 'done' && i.result && i.zip !== false);
   790→  if (!done.length) return;
   791→  const { toast } = await import('sonner');
   792→  const t = toast.loading(`Packing ${done.length} image${done.length > 1 ? 's' : ''}…`);
   793→  try {
   794→    const JSZip = (await import('jszip')).default;
   795→    const zip = new JSZip();
   796→    const used = new Set<string>();
   797→    for (const item of done) {
   798→      const r = item.result!;
   799→      let filename = resultFilename(item.name, r.scale, r.format);
   800→      let i = 1;
   801→      while (used.has(filename)) {
   802→        filename = resultFilename(item.name, r.scale, r.format).replace(/(\.\w+)$/, `-${i++}$1`);
   803→      }
   804→      used.add(filename);
   805→      const blob = await (await fetch(r.url)).blob();
   806→      zip.file(filename, blob);
   807→    }
   808→    const out = await zip.generateAsync({ type: 'blob' });
   809→    const stamp = new Date().toISOString().slice(0, 16).replace(/[-:T]/g, '');
   810→    downloadBlob(out, `pixelforge-upscaled-${stamp}.zip`);
   811→    toast.success(`ZIP ready — ${formatBytes(out.size)}`, { id: t });
   812→  } catch (err) {
   813→    console.error(err);
   814→    toast.error('Could not build the ZIP archive', { id: t });
   815→  }
   816→}
