    99→/**
   100→ * Live AI patch mosaic — every cell is one 128px patch the neural engine has
   101→ * finished. Giant grids (>220 patches) fall back to a sampled 16×12 mosaic so
   102→ * DOM size stays bounded while the look stays accurate.
   103→ */
   104→function PatchGrid({ patch }: { patch: NonNullable<QueueItem['patch']> }) {
   105→  const total = patch.cols * patch.rows;
   106→  if (total <= 1) return null;
   107→  const sampled = total > 220 ? { cols: 16, rows: 12 } : { cols: patch.cols, rows: patch.rows };
   108→  const shownTotal = sampled.cols * sampled.rows;
   109→  const doneShown = Math.min(shownTotal, Math.round((patch.done / total) * shownTotal));
   110→  const cells: React.ReactNode[] = [];
   111→  for (let i = 0; i < shownTotal; i++) {
   112→    const isDone = i < doneShown;
   113→    cells.push(
   114→      <span
   115→        key={i}
   116→        aria-hidden
   117→        className={cn(
   118→          'rounded-[1px]',
   119→          isDone ? 'pf-patch-cell bg-primary/80 shadow-[0_0_4px] shadow-primary/40' : 'bg-foreground/10'
   120→        )}
   121→      />
   122→    );
   123→  }
   124→  return (
   125→    <div
   126→      className="pointer-events-none absolute inset-1.5 grid gap-[2px] p-0"
   127→      style={{
   128→        gridTemplateColumns: `repeat(${sampled.cols}, minmax(0,1fr))`,
   129→        gridTemplateRows: `repeat(${sampled.rows}, minmax(0,1fr))`,
   130→      }}
   131→    >
   132→      {cells}
   133→    </div>
   134→  );
   135→}

   275→function buildMeta(item: QueueItem): string {
   276→  const r = item.result;
   277→  const meta: string[] = [];
   278→  if (item.w) meta.push(`${item.w}×${item.h}`);
   279→  if (r) meta.push(`→ ${r.w}×${r.h}`);
   280→  if (item.sizeIn) meta.push(formatBytes(item.sizeIn));
   281→  if (r && r.size > 0) meta.push(`→ ${formatBytes(r.size)}`);
   282→  if (r && r.ms > 0) meta.push(formatDuration(r.ms));
   283→  return meta.join(' · ');
   284→}

   496→export function Queue() {
   497→  const items = useStore((s) => s.items);
   498→  const busy = useStore((s) => s.busy);
   499→  const paused = useStore((s) => s.paused);
   500→  const togglePause = useStore((s) => s.togglePause);
   501→  const clearFinished = useStore((s) => s.clearFinished);
   502→  const clearAll = useStore((s) => s.clearAll);
   503→  const backend = useStore((s) => s.backend);
   504→  const gpu = useStore((s) => s.gpu);
   505→  const totals = useStore((s) => s.totals);
   506→  const settings = useStore((s) => s.settings);
   507→  const setSettings = useStore((s) => s.setSettings);
   508→  const view = settings.queueView;
   509→
   510→  const doneCount = items.filter((i) => i.status === 'done').length;
   511→  const queuedCount = items.filter((i) => i.status === 'queued').length;
   512→  const zipCount = items.filter((i) => i.status === 'done' && i.result && i.zip !== false).length;
   513→  const active = items.find((i) => i.status === 'processing');
   514→  const overall =
   515→    items.length === 0
   516→      ? 0
   517→      : (doneCount + (active ? active.progress : 0)) / items.length;
   518→
   519→  const hasDone = doneCount > 0;
   520→  const hasQueued = queuedCount > 0;
   521→
   522→  // rough ETA from the session average — only meaningful once ≥1 image finished
   523→  const avgMs = totals.images > 0 ? totals.ms / totals.images : 0;
   524→  const eta = busy && !paused && hasQueued && avgMs > 0 ? avgMs * (queuedCount + (active ? 1 - active.progress : 0)) : 0;
   525→
   526→  return (
   527→    <div className="space-y-3">
   528→      {/* batch bar */}
   529→      <div
   530→        className={cn(
   531→          'rounded-xl border bg-card/60 p-3 transition-colors',
   532→          paused && busy && 'border-amber-500/40 bg-amber-500/5'
   533→        )}
   534→      >
   535→        <div className="flex flex-wrap items-center gap-2">
   536→          <div className="flex flex-wrap items-center gap-2">
   537→            {busy && !paused && (
   538→              <Button size="sm" variant="outline" className="h-8 gap-1.5" onClick={togglePause} disabled={!hasQueued}>
   539→                <Pause className="h-3.5 w-3.5" aria-hidden />
   540→                Pause
   541→              </Button>
   542→            )}
   543→            {paused && (
   544→              <Button
   545→                size="sm"
   546→                variant="outline"
   547→                className="h-8 gap-1.5"
   548→                onClick={togglePause}
   549→                disabled={!hasQueued}
   550→              >
   551→                <Play className="h-3.5 w-3.5" aria-hidden />
   552→                Resume
   553→              </Button>
   554→            )}
   555→            <TooltipProvider delayDuration={200}>
   556→              <Tooltip>
   557→                <TooltipTrigger asChild>
   558→                  <Button
   559→                    size="sm"
   560→                    variant="outline"
   561→                    className="h-8 gap-1.5"
   562→                    onClick={() => void downloadAllAsZip()}
   563→                    disabled={zipCount === 0}
   564→                  >
   565→                    <FileArchive className="h-3.5 w-3.5" aria-hidden />
   566→                    ZIP all
   567→                    <span className="font-mono text-[10px] text-muted-foreground">
   568→                      ({zipCount}
   569→                      {hasDone && zipCount !== doneCount ? `/${doneCount}` : ''})
   570→                    </span>
   571→                  </Button>
   572→                </TooltipTrigger>
   573→                {hasDone && zipCount !== doneCount && (
   574→                  <TooltipContent>Deselect items with the ZIP checkboxes to skip them</TooltipContent>
   575→                )}
   576→              </Tooltip>
   577→            </TooltipProvider>
   578→            <Button size="sm" variant="ghost" className="h-8" onClick={clearFinished} disabled={!items.some((i) => i.status !== 'queued' && i.status !== 'processing')}>
   579→              Clear finished
   580→            </Button>
   581→            <Button size="sm" variant="ghost" className="h-8 text-muted-foreground hover:text-destructive" onClick={clearAll}>
   582→              <Trash2 className="h-3.5 w-3.5" aria-hidden />
   583→              Clear all
   584→            </Button>
   585→          </div>
   586→
   587→          <div className="ml-auto flex items-center gap-2">
   588→            {eta > 0 && (
   589→              <Badge
   590→                variant="outline"
   591→                className="gap-1 font-mono text-[10px] tabular-nums text-muted-foreground"
   592→                title={`Based on the ${formatDuration(avgMs)} average of this session`}
   593→              >
   594→                <Timer className="h-3 w-3" aria-hidden />
   595→                ~{formatDuration(eta)} left
   596→              </Badge>
   597→            )}
   598→            {backend && (
   599→              <Badge
   600→                variant="outline"
   601→                className="gap-1 font-mono text-[10px] text-muted-foreground"
   602→                title={gpu ? `GPU: ${gpu}` : undefined}
   603→              >
   604→                <Cpu className="h-3 w-3" aria-hidden />
   605→                {backend === 'webgl' ? 'WebGL' : backend.toUpperCase()}
   606→              </Badge>
   607→            )}
   608→            <Badge variant="outline" className="font-mono text-[10px] tabular-nums text-muted-foreground">
   609→              {doneCount}/{items.length} done
   610→              {totals.ms > 0 && ` · ${formatDuration(totals.ms)} total`}
   611→            </Badge>
   612→            {/* shortcuts + view switch */}
   613→            <div className="flex items-center gap-1">
   614→              <TooltipProvider delayDuration={200}>
   615→                <Tooltip>
   616→                  <TooltipTrigger asChild>
   617→                    <button
   618→                      type="button"
   619→                      aria-label="Keyboard shortcuts"
   620→                      title="Keyboard shortcuts (?)"
   621→                      onClick={openShortcuts}
   622→                      className="flex h-7 w-7 items-center justify-center rounded-md text-muted-foreground transition-colors hover:bg-muted hover:text-foreground"
   623→                    >
   624→                      <Keyboard className="h-3.5 w-3.5" aria-hidden />
   625→                    </button>
   626→                  </TooltipTrigger>
   627→                  <TooltipContent>Keyboard shortcuts</TooltipContent>
   628→                </Tooltip>
   629→              </TooltipProvider>
   630→              <div className="flex items-center rounded-lg border p-0.5" role="group" aria-label="Queue layout">
   631→                <button
   632→                  type="button"
   633→                  aria-label="List view"
   634→                  aria-pressed={view === 'list'}
   635→                  title="List view (V)"
   636→                  onClick={() => setSettings({ queueView: 'list' })}
   637→                  className={cn(
   638→                    'flex h-7 w-8 items-center justify-center rounded-md transition-colors',
   639→                    view === 'list' ? 'bg-primary/15 text-primary' : 'text-muted-foreground hover:bg-muted hover:text-foreground'
   640→                  )}
   641→                >
   642→                  <LayoutList className="h-3.5 w-3.5" aria-hidden />
   643→                </button>
   644→                <button
   645→                  type="button"
   646→                  aria-label="Grid view"
   647→                  aria-pressed={view === 'grid'}
   648→                  title="Grid view (V)"
   649→                  onClick={() => setSettings({ queueView: 'grid' })}
   650→                  className={cn(
   651→                    'flex h-7 w-8 items-center justify-center rounded-md transition-colors',
   652→                    view === 'grid' ? 'bg-primary/15 text-primary' : 'text-muted-foreground hover:bg-muted hover:text-foreground'
   653→                  )}
   654→                >
   655→                  <LayoutGrid className="h-3.5 w-3.5" aria-hidden />
   656→                </button>
   657→              </div>
   658→            </div>
   659→          </div>
   660→        </div>
   661→
   662→        <div className="mt-2.5 flex items-center gap-2">
   663→          <Progress value={overall * 100} className="h-1.5" aria-label="Overall batch progress" />
   664→          <span className="w-9 shrink-0 text-right font-mono text-[10px] tabular-nums text-muted-foreground">
   665→            {Math.round(overall * 100)}%
   666→          </span>
   667→        </div>
   668→        {paused && busy && (
   669→          <p className="mt-1.5 text-[11px] text-amber-500">
   670→            Paused — the current image will finish, queued ones wait.
   671→          </p>
   672→        )}
   673→      </div>
   674→
   675→      {/* items */}
   676→      {view === 'list' ? (
   677→        <ul className="max-h-[52vh] space-y-2 overflow-y-auto pr-1 fancy-scroll" role="list">
   678→          {items.map((item, i) => (
   679→            <QueueItemRow key={item.id} item={item} index={i} />
   680→          ))}
   681→        </ul>
   682→      ) : (
   683→        <ul className="max-h-[58vh] grid grid-cols-2 gap-2.5 overflow-y-auto pr-1 fancy-scroll sm:grid-cols-3 xl:grid-cols-4" role="list">
   684→          {items.map((item, i) => (
   685→            <QueueCard key={item.id} item={item} index={i} />
   686→          ))}
   687→        </ul>
   688→      )}
   689→    </div>
   690→  );
   691→}
